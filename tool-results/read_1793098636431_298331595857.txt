    30→export type GlassMaterial = 'auto' | 'frost' | 'clear';
    31→export type GlassProfile = 'squircle' | 'convex' | 'rim';
    32→
    33→/**
    34→ * Squircle corners read "less round" than circular ones at the same radius —
    35→ * the superellipse hugs the corner and keeps its curvature away from the
    36→ * edges — so surfaces looked tighter than the circular-corner releases at
    37→ * identical numbers. Where the squircle actually renders, the declared
    38→ * radius is inflated to restore the perceived corner size; elsewhere the
    39→ * value passes through and the look matches older releases exactly. The
    40→ * inflated radius feeds both --lq-radius and the displacement map, so clip
    41→ * and optics stay on one geometry.
    42→ */
    43→const SQUIRCLE_RADIUS_SCALE = 1.4;
    44→const supportsSquircle =
    45→  typeof window !== 'undefined' &&
    46→  typeof CSS !== 'undefined' &&
    47→  CSS.supports('corner-shape', 'squircle');
    48→
    49→/**
    50→ * Refraction magnitude along the bezel, indexed by normalized depth from the
    51→ * outer edge (0 = edge, 1 = inner end of the bezel). The physical profiles
    52→ * model a glass slab whose top surface follows a height function h(t): the
    53→ * surface slope gives the incident angle, Snell's law (n = 1.5) the bend, and
    54→ * the remaining glass depth the lateral shift — giving the sharp peak at the
    55→ * rim and smooth falloff of real thick glass. 'rim' is the cheap stylized
    56→ * falloff kept for comparison.
    57→ */
    58→const LUT_SIZE = 128;
    59→interface SurfaceLUTs {
    60→  /** Refraction displacement magnitude, normalized to [0, 1]. */
    61→  mag: Float32Array;
    62→  /** Surface slope dh/dd × thickness — drives the 3D normal for lighting. */
    63→  slope: Float32Array;
    64→}
    65→const lutCache = new Map<GlassProfile, SurfaceLUTs>();
    66→
    67→function surfaceLUTs(profile: GlassProfile): SurfaceLUTs {
    68→  const cached = lutCache.get(profile);
    69→  if (cached) return cached;
    70→
    71→  const mag = new Float32Array(LUT_SIZE);
    72→  const slope = new Float32Array(LUT_SIZE);
    73→  const n = 1.5; // refractive index of glass
    74→  const T = 0.6; // slab thickness relative to bezel width
    75→  const h =
    76→    profile === 'squircle'
    77→      ? (t: number) => Math.pow(1 - Math.pow(1 - t, 4), 0.25)
    78→      : profile === 'convex'
    79→        ? (t: number) => Math.sqrt(1 - (1 - t) * (1 - t))
    80→        : (t: number) => 1 - (1 - t) * (1 - t); // 'rim' pseudo-height
    81→  const eps = 1 / 1024;
    82→  let max = 0;
    83→  for (let i = 0; i < LUT_SIZE; i++) {
    84→    const t = Math.max(i / (LUT_SIZE - 1), eps);
    85→    const hi = Math.min(t + eps, 1);
    86→    const lo = Math.max(t - eps, 0);
    87→    slope[i] = ((h(hi) - h(lo)) / (hi - lo)) * T;
    88→    if (profile === 'rim') {
    89→      mag[i] = (1 - t) * (1 - t);
    90→    } else {
    91→      const thetaI = Math.atan(Math.abs(slope[i]));
    92→      const delta = thetaI - Math.asin(Math.sin(thetaI) / n);
    93→      mag[i] = h(t) * T * Math.tan(delta);
    94→      max = Math.max(max, mag[i]);
    95→    }
    96→  }
    97→  if (max > 0) for (let i = 0; i < LUT_SIZE; i++) mag[i] /= max;
    98→  const luts = { mag, slope };
    99→  lutCache.set(profile, luts);
   100→  return luts;
   101→}

   103→/**
   104→ * Rim lighting is derived from the surface normal: a directional light from
   105→ * above and its counterpart from below catch the top and bottom arcs — a
   106→ * straight edge lights evenly along its whole run — while the left/right
   107→ * flanks fall into a thin shade. What keeps this from reading as a bevelled
   108→ * button is the shaping: the highlight lives in a ~1px stroke hugging the
   109→ * boundary plus a few px of diffuse glow, not a wide band across the bezel.
   110→ * (The earlier positional-azimuth model lit straight edges unevenly, which is
   111→ * the single most visible tell against real glass.)
   112→ */
   113→const LIGHT_SPREAD = 1.08; // radians off vertical the top/bottom highlight reaches
   114→const DIFFUSE_SPREAD = LIGHT_SPREAD * 0.65;
   115→const SIDE_SPREAD = 0.98; // radians off horizontal the side shade reaches
   116→
   117→const sat = (x: number) => (x < 0 ? 0 : x > 1 ? 1 : x);
   118→
   119→interface SurfaceField {
   120→  /** Signed distance to the outline, negative inside. */
   121→  sd: number;
   122→  nx: number;
   123→  ny: number;
   124→}
   125→
   126→/**
   127→ * Signed distance and outward normal for the surface outline: straight edges
   128→ * joined by superellipse corners, |x|⁴ + |y|⁴ = r⁴ — the curve CSS
   129→ * `corner-shape: squircle` draws, so the refraction map, the baked lighting
   130→ * and the CSS clip agree on one geometry. The corner distance is a
   131→ * first-order estimate (field / |gradient|): exact at the curve, drifting
   132→ * only well inside, where nothing reads absolute distance anymore.
   133→ */
   134→function surfaceField(
   135→  px: number,
   136→  py: number,
   137→  bx: number,
   138→  by: number,
   139→  r: number,
   140→  out: SurfaceField,
   141→) {
   142→  const qx = Math.abs(px) - bx;
   143→  const qy = Math.abs(py) - by;
   144→  if (qx > 0 && qy > 0) {
   145→    const x3 = qx * qx * qx;
   146→    const y3 = qy * qy * qy;
   147→    const q = Math.pow(qx * x3 + qy * y3, 0.25);
   148→    const len = Math.hypot(x3, y3) || 1;
   149→    out.sd = ((q - r) * q * q * q) / len;
   150→    out.nx = (Math.sign(px) * x3) / len;
   151→    out.ny = (Math.sign(py) * y3) / len;
   152→  } else {
   153→    const ox = Math.max(qx, 0);
   154→    const oy = Math.max(qy, 0);
   155→    out.sd = Math.hypot(ox, oy) + Math.min(Math.max(qx, qy), 0) - r;
   156→    if (qx > qy) {
   157→      out.nx = Math.sign(px);
   158→      out.ny = 0;
   159→    } else {
   160→      out.nx = 0;
   161→      out.ny = Math.sign(py);
   162→    }
   163→  }
   164→}
   165→
   166→/**
   167→ * Displacement map for the lens effect, generated per-pixel on a canvas.
   168→ * R encodes horizontal displacement, B vertical; 128 is neutral. A
   169→ * rounded-rect signed distance field gives depth + outward normal per pixel;
   170→ * the LUT above gives the magnitude. Pixels sample toward the center
   171→ * (convex-lens edge magnification) — sampling outward would hit Chromium's
   172→ * backdrop edge-clamp and smear instead of refract.
   173→ *
   174→ * (Canvas instead of an SVG data-URI: feImage rasterizes SVG images with CSS
   175→ * features like mix-blend-mode disabled, which silently corrupts
   176→ * gradient-composited maps.)
   177→ */
   178→/**
   179→ * Maps are cached module-wide: popups remount at identical sizes on every
   180→ * open, so a reopen costs a Map lookup instead of a canvas render. Large
   181→ * displacement maps render at half resolution — displacement vectors are a
   182→ * smooth field, so feImage stretching is invisible, and generation + PNG
   183→ * decode get ~4× cheaper (less pop-in latency). The specular always renders
   184→ * at full resolution: its rim is a ~1px stroke, which is exactly the feature
   185→ * stretching would soften.
   186→ *
   187→ * Bounded LRU: a surface that animates its box (an accordion panel expanding)
   188→ * walks through a distinct size every frame, so an unbounded cache would grow
   189→ * a PNG set per intermediate height and never release them.
   190→ */
   191→const IMAGE_CACHE_MAX = 48;
   192→const imageCache = new Map<string, { map: string; specular: string; specularDark: string }>();
   193→
   194→function cacheGet(key: string) {
   195→  const hit = imageCache.get(key);
   196→  if (!hit) return undefined;
   197→  // Re-insert to mark most-recently-used (Map preserves insertion order).
   198→  imageCache.delete(key);
   199→  imageCache.set(key, hit);
   200→  return hit;
   201→}
   202→
   203→function cacheSet(key: string, value: { map: string; specular: string; specularDark: string }) {
   204→  imageCache.set(key, value);
   205→  if (imageCache.size > IMAGE_CACHE_MAX) {
   206→    imageCache.delete(imageCache.keys().next().value!);
   207→  }
   208→}

   210→function glassImages(
   211→  fullW: number,
   212→  fullH: number,
   213→  fullRadius: number,
   214→  fullBezel: number,
   215→  profile: GlassProfile,
   216→) {
   217→  const key = `${fullW}x${fullH}r${fullRadius}b${fullBezel}${profile}`;
   218→  const cached = cacheGet(key);
   219→  if (cached) return cached;
   220→
   221→  const scale = fullW * fullH > 32000 ? 0.5 : 1;
   222→  const w = Math.ceil(fullW * scale);
   223→  const h = Math.ceil(fullH * scale);
   224→  const radius = fullRadius * scale;
   225→  const bezel = fullBezel * scale;
   226→
   227→  const canvas = document.createElement('canvas');
   228→  canvas.width = w;
   229→  canvas.height = h;
   230→  const ctx = canvas.getContext('2d')!;
   231→  const image = ctx.createImageData(w, h);
   232→  const data = image.data;
   233→
   234→  const { mag: lut } = surfaceLUTs(profile);
   235→  const field: SurfaceField = { sd: 0, nx: 0, ny: 0 };
   236→
   237→  const r = Math.min(radius, w / 2, h / 2);
   238→  const bx = w / 2 - r;
   239→  const by = h / 2 - r;
   240→
   241→  for (let y = 0; y < h; y++) {
   242→    for (let x = 0; x < w; x++) {
   243→      const px = x + 0.5 - w / 2;
   244→      const py = y + 0.5 - h / 2;
   245→      surfaceField(px, py, bx, by, r, field);
   246→      const depth = -field.sd; // distance from the edge, measured inward
   247→
   248→      const d = depth / bezel;
   249→      const inRim = d < 1 && d >= 0;
   250→      const idx = inRim ? Math.min(Math.round(d * (LUT_SIZE - 1)), LUT_SIZE - 1) : 0;
   251→      const mag = inRim ? lut[idx] : 0;
   252→
   253→      const i = (y * w + x) * 4;
   254→      data[i] = Math.round(128 - field.nx * mag * 127); // R → x displacement
   255→      data[i + 1] = 128;
   256→      data[i + 2] = Math.round(128 - field.ny * mag * 127); // B → y displacement
   257→      data[i + 3] = 255;
   258→    }
   259→  }
   260→  ctx.putImageData(image, 0, 0);
   261→
   262→  /**
   263→   * Specular overlay, one image per appearance. Unpremultiplied compose of
   264→   * two features, both driven by the outline normal and distance:
   265→   *
   266→   * - highlight — a crisp stroke hugging the boundary where the normal faces
   267→   *   up or down (times a soft compression so it never clips), plus a wider
   268→   *   diffuse glow fading over a few px into the surface;
   269→   * - shade — a hairline just inside the left/right flanks, the dark
   270→   *   contact edge that makes the bright arcs read as a lit rim rather than
   271→   *   a painted-on white line.
   272→   *
   273→   * Dark appearance dims the highlight and deepens the shade; the two PNGs
   274→   * are swapped by CSS (see glass.css) so the appearance change needs no JS.
   275→   */
   276→  const specW = fullW;
   277→  const specH = fullH;
   278→  const specR = Math.min(fullRadius, specW / 2, specH / 2);
   279→  const specBx = specW / 2 - specR;
   280→  const specBy = specH / 2 - specR;
   281→
   282→  const makeSpecCanvas = () => {
   283→    const c = document.createElement('canvas');
   284→    c.width = specW;
   285→    c.height = specH;
   286→    return c;
   287→  };
   288→  const lightCanvas = makeSpecCanvas();
   289→  const darkCanvas = makeSpecCanvas();
   290→  const lightCtx = lightCanvas.getContext('2d')!;
   291→  const darkCtx = darkCanvas.getContext('2d')!;
   292→  const lightImage = lightCtx.createImageData(specW, specH);
   293→  const darkImage = darkCtx.createImageData(specW, specH);
   294→
   295→  const cosLight = Math.cos(LIGHT_SPREAD);
   296→  const cosDiffuse = Math.cos(DIFFUSE_SPREAD);
   297→  const cosSide = Math.cos(SIDE_SPREAD);
   298→  // The diffuse glow reaches a few px into the surface, growing gently with
   299→  // the bezel so wide rims keep a visible inner falloff.
   300→  const reach = 4.5 + fullBezel * 0.15;
   301→
   302→  const compose = (
   303→    out: Uint8ClampedArray,
   304→    i: number,
   305→    specAmount: number,
   306→    diffuse: number,
   307→    shadeBase: number,
   308→    specScale: number,
   309→    diffuseScale: number,
   310→    shadeStrength: number,
   311→  ) => {
   312→    const highlight = sat(specAmount * specScale + diffuse * diffuseScale);
   313→    const shade = shadeBase * shadeStrength;
   314→    const alpha = highlight + shade * (1 - highlight);
   315→    if (alpha <= 0) return;
   316→    // Light over a dark rim, unpremultiplied for the PNG encoder.
   317→    const value = Math.min((specScale * highlight) / alpha, 1);
   318→    const v = Math.round(value * 255);
   319→    out[i] = v;
   320→    out[i + 1] = v;
   321→    out[i + 2] = v;
   322→    out[i + 3] = Math.round(Math.min(alpha, 1) * 255);
   323→  };
   324→
   325→  for (let y = 0; y < specH; y++) {
   326→    for (let x = 0; x < specW; x++) {
   327→      const px = x + 0.5 - specW / 2;
   328→      const py = y + 0.5 - specH / 2;
   329→      surfaceField(px, py, specBx, specBy, specR, field);
   330→      const sd = field.sd;
   331→      // Everything lives within a hair of the boundary; skip the bulk early.
   332→      if (sd > 1 || sd < -(reach + 1)) continue;
   333→
   334→      const coverage = sat(0.5 - sd);
   335→      const vertical = Math.abs(field.ny);
   336→      const directional = Math.pow(sat((vertical - cosLight) / (1 - cosLight)), 1.25);
   337→      const diffuseDir = Math.pow(sat((vertical - cosDiffuse) / (1 - cosDiffuse)), 1.25);
   338→      const stroke = sat(sd + 1.4) * sat(0.5 - sd);
   339→      let specAmount = directional * stroke;
   340→      specAmount = specAmount / (1 + 0.5 * (1 - specAmount));
   341→      const glow = 1 - sat(-sd / reach);
   342→      const diffuse = diffuseDir * 0.18 * glow * glow * coverage;
   343→
   344→      // Side shade: same hairline band, weighted toward horizontal normals.
   345→      const wL = sat((field.nx - cosSide) / (1 - cosSide)) * stroke;
   346→      const wR = sat((-field.nx - cosSide) / (1 - cosSide)) * stroke;
   347→      const shadeBase = wL / (1 + 0.4 * (1 - wL)) + wR / (1 + 0.4 * (1 - wR));
   348→
   349→      const i = (y * specW + x) * 4;
   350→      compose(lightImage.data, i, specAmount, diffuse, shadeBase, 1, 1, 0.48);
   351→      compose(darkImage.data, i, specAmount, diffuse, shadeBase, 0.82, 0.4, 0.6);
   352→    }
   353→  }
   354→  lightCtx.putImageData(lightImage, 0, 0);
   355→  darkCtx.putImageData(darkImage, 0, 0);
   356→
   357→  const result = {
   358→    map: canvas.toDataURL(),
   359→    specular: lightCanvas.toDataURL(),
   360→    specularDark: darkCanvas.toDataURL(),
   361→  };
   362→  cacheSet(key, result);
   363→  return result;
   364→}

   366→export interface LiquiGlassProps extends React.HTMLAttributes<HTMLDivElement> {
   367→  /**
   368→   * Rendering tier — the perf/quality trade-off, exposed so apps (or their
   369→   * users) can downgrade:
   370→   * - 'auto'  — true refraction where supported, frosted glass elsewhere
   371→   * - 'frost' — always frosted (blur + saturate); no SVG filter, no canvas map
   372→   * - 'clear' — tint and rim only, no backdrop-filter at all (cheapest)
   373→   */
   374→  material?: GlassMaterial;
   375→  /** Lens shape of the refracting rim. 'squircle' is the most glass-like. */
   376→  profile?: GlassProfile;
   377→  /**
   378→   * Corner radius of the glass surface in px.
   379→   *
   380→   * This and the three below are the *geometric* dials: they scale with the
   381→   * surface, so a theme multiplies them (`radiusScale` and friends) rather than
   382→   * replacing them. See `LiquiThemeProvider`.
   383→   */
   384→  radius?: number;
   385→  /** Backdrop blur in px (refraction tier; frost uses 4×). */
   386→  blur?: number;
   387→  /** Strength of the edge refraction in px (feDisplacementMap scale). */
   388→  refraction?: number;
   389→  /** Width in px of the refracting rim at the edge of the surface. */
   390→  bezel?: number;
   391→  /**
   392→   * Chromatic aberration: 0 disables (single displacement pass), >0 splits
   393→   * R/G/B into three displacement passes at ±dispersion relative scale.
   394→   * Costs roughly 3× filter work — keep 0 where perf matters.
   395→   */
   396→  dispersion?: number;
   397→  /**
   398→   * Opacity of the normal-lit specular layer (0 disables). Only rendered in
   399→   * the refraction tier; frost/clear keep the cheap inset-shadow shine.
   400→   */
   401→  specular?: number;
   402→  /**
   403→   * Material density, 0–1, mirroring Apple's Liquid Glass variants:
   404→   * 0 ≈ "clear" (transparent, for media-rich backdrops with their own
   405→   * dimming), 1 ≈ "regular" (adaptive frosted tint that carries legibility).
   406→   * Interpolates tint opacity and adds up to 14px of blur on top of `blur`.
   407→   */
   408→  frost?: number;
   409→  /** Extra saturation pushed through the glass. */
   410→  saturation?: number;
   411→  /** Depth level: raises tint opacity + shadow. */
   412→  elevated?: boolean;
   413→  /**
   414→   * Classes for the content wrapper that sits above every glass layer.
   415→   *
   416→   * Padding, flex layout and typography belong to the *component* built on the
   417→   * glass, not to the glass — but that wrapper is rendered here, so a consumer
   418→   * styling it from outside would need a descendant selector
   419→   * (`[&>.liqui-glass__content]:…`), which utility classes express badly. This
   420→   * prop hands the wrapper over directly so a registry component can stay pure
   421→   * Tailwind.
   422→   */
   423→  contentClassName?: string;
   424→}

   426→/**
   427→ * LiquiGlass — the base surface primitive of the liqui component library.
   428→ * Layers: refracting backdrop → tint → specular rim → content.
   429→ */
   430→export const LiquiGlass = React.forwardRef<HTMLDivElement, LiquiGlassProps>(
   431→  function LiquiGlass(props, forwardedRef) {
   432→    // Theme defaults fill in only where the surface said nothing. A component
   433→    // that hardcodes `frost: 0.6` because its design needs it keeps that value
   434→    // under every theme; the theme owns the dials nobody claimed.
   435→    const theme = useGlassTheme();
   436→    const {
   437→      material = theme.material,
   438→      profile = theme.profile,
   439→      radius: ownRadius = 16,
   440→      blur: ownBlur = 1,
   441→      refraction: ownRefraction = 140,
   442→      bezel: ownBezel = 26,
   443→      dispersion = theme.dispersion,
   444→      specular = theme.specular,
   445→      frost = theme.frost,
   446→      saturation = theme.saturation,
   447→      elevated = false,
   448→      className,
   449→      contentClassName,
   450→      style,
   451→      children,
   452→      ...rest
   453→    } = props;
   454→
   455→    // `supportsRefraction` sniffs the UA, so it is false during SSR and true in
   456→    // Chromium — reading it directly at render time would make the server emit
   457→    // the frost tier and the client's first render emit the refract tier, which
   458→    // is a hydration mismatch React explicitly won't patch up. Start every
   459→    // render pass in the tier the server can also produce, then upgrade in a
   460→    // layout effect: that runs after hydration commits but before paint, so
   461→    // there is no visible frost flash. `squircleActive` follows the same
   462→    // pattern: the radius compensation only applies once the client knows
   463→    // corner-shape will actually render, and must not differ from the server
   464→    // markup on the hydration pass.
   465→    const [refractionAllowed, setRefractionAllowed] = React.useState(false);
   466→    const [squircleActive, setSquircleActive] = React.useState(false);
   467→    React.useLayoutEffect(() => {
   468→      if (supportsRefraction) setRefractionAllowed(true);
   469→      if (supportsSquircle) setSquircleActive(true);
   470→    }, []);
   471→
   472→    // Geometry is scaled, not replaced — see theme.tsx. Rounding is not
   473→    // cosmetic: radius and bezel key the displacement-map cache and refraction
   474→    // keys the filter registry, so a multiplier dragged through 0.9137× would
   475→    // otherwise mint a fresh canvas render and a fresh <filter> node per frame.
   476→    // Blur only reaches a CSS string, but is rounded for the same reason in
   477→    // reverse — an unrounded one churns the style attribute every frame.
   478→    const radius = Math.round(
   479→      ownRadius * theme.radiusScale * (squircleActive ? SQUIRCLE_RADIUS_SCALE : 1),
   480→    );
   481→    const blur = Math.round(ownBlur * theme.blurScale * 100) / 100;
   482→    const refraction = Math.round(ownRefraction * theme.refractionScale);
   483→    // A zero-width bezel divides through in the map generator (`depth / bezel`),
   484→    // turning the whole surface into NaN displacement — a fully smeared box.
   485→    const bezel = Math.max(1, Math.round(ownBezel * theme.bezelScale));
   486→
   487→    const tier: 'refract' | 'frost' | 'clear' =
   488→      material === 'clear'
   489→        ? 'clear'
   490→        : material === 'frost'
   491→          ? 'frost'
   492→          : refractionAllowed
   493→            ? 'refract'
   494→            : 'frost';
   495→
   496→    const localRef = React.useRef<HTMLDivElement | null>(null);
   497→    const [size, setSize] = React.useState<{ w: number; h: number } | null>(null);
   498→
   499→    // The empty dependency array is load-bearing. Without it this runs on every
   500→    // render, and each run detaches the previous handle before attaching the
   501→    // new one — which, when the forwarded ref is a *callback* ref, means calling
   502→    // it with `null` and then with the element again, on every render.
   503→    //
   504→    // Base UI hands exactly such a callback ref to any part that belongs to a
   505→    // composite list (a slider thumb, a tab, a menu item), and its cleanup
   506→    // unregisters the item and schedules a state update. Re-render, detach,
   507→    // unregister, re-render: "Maximum update depth exceeded", from a component
   508→    // that is only sitting there. Base UI 1.7 is where the unregister path
   509→    // started scheduling that update, so that is where this surfaced, but the
   510→    // unstable handle was always the defect.
   511→    //
   512→    // The handle is a getter over a ref that outlives every render, so there is
   513→    // nothing for it to depend on.
   514→    React.useImperativeHandle(forwardedRef, () => localRef.current as HTMLDivElement, []);
   515→
   516→    React.useLayoutEffect(() => {
   517→      if (tier !== 'refract' || !localRef.current) return;
   518→      const el = localRef.current;
   519→      const apply = (w: number, h: number) => {
   520→        if (w > 0 && h > 0) {
   521→          setSize((prev) => (prev && prev.w === w && prev.h === h ? prev : { w, h }));
   522→        }
   523→      };
   524→      // Measure synchronously before first paint so the filter is present on
   525→      // the very first frame instead of popping in a few frames later.
   526→      // offsetWidth/contentRect are layout sizes, unaffected by CSS
   527→      // transforms — measuring the bounding rect during the scale(0.85) open
   528→      // animation would bake a permanently undersized map.
   529→      apply(el.offsetWidth, el.offsetHeight);
   530→      const observer = new ResizeObserver((entries) => {
   531→        const rect = entries[0].contentRect;
   532→        apply(Math.round(rect.width), Math.round(rect.height));
   533→      });
   534→      observer.observe(el);
   535→      return () => observer.disconnect();
   536→    }, [tier]);
   537→
   538→    const refractionReady = tier === 'refract' && size !== null;
   539→    // Tone response of the material: lift the darkest backdrop tones and roll
   540→    // off the brightest, so refracted content sits in a narrower band and the
   541→    // foreground keeps contrast against it. Clear glass (frost 0) needs the
   542→    // full curve — nothing else quiets the backdrop; heavy frost already
   543→    // carries legibility through blur + tint, so the curve relaxes. Rounded to
   544→    // 3 decimals because these key the filter registry (see the geometry
   545→    // rounding note above).
   546→    const toneLift = Math.round(180 * (1 - 0.45 * frost)) / 1000;
   547→    const toneRoll = Math.round(140 * (1 - 0.45 * frost)) / 1000;
   548→    // Filters live in a global, never-unmounted registry so a remounted
   549→    // popup references an already-decoded filter (see filterRegistry.ts).
   550→    const glassRefs = React.useMemo(() => {
   551→      if (!refractionReady) return null;
   552→      const images = glassImages(size!.w, size!.h, radius, bezel, profile);
   553→      prewarmImage(images.specular);
   554→      prewarmImage(images.specularDark);
   555→      const { id: filterId, cold } = ensureFilter({
   556→        w: size!.w,
   557→        h: size!.h,
   558→        mapHref: images.map,
   559→        refraction,
   560→        dispersion,
   561→        toneLift,
   562→        toneRoll,
   563→      });
   564→      return { images, filterId, cold };
   565→    }, [refractionReady, size, radius, bezel, profile, refraction, dispersion, toneLift, toneRoll]);
   566→
   567→    // The masking fade-in is only for a cold filter (feImage still decoding).
   568→    // It must also be REMOVED after it plays: under keepMounted a closed
   569→    // popup is display:none, and re-display restarts any CSS animation still
   570→    // on the element — which would re-fade refraction on every open.
   571→    const [fadeDone, setFadeDone] = React.useState(false);
   572→    const needsFade = (glassRefs?.cold ?? false) && !fadeDone;
   573→    React.useEffect(() => {
   574→      if (!glassRefs?.cold || fadeDone) return;
   575→      const timer = setTimeout(() => setFadeDone(true), 250);
   576→      return () => clearTimeout(timer);
   577→    }, [glassRefs?.cold, fadeDone]);
   578→
   579→    // frost interpolates toward Apple's "regular" variant: more blur + tint.
   580→    const effectiveBlur = blur + frost * 14;
   581→    // Blur and refraction live on SEPARATE backdrop layers. While feImage
   582→    // decodes its data URL, Chromium treats the whole
   583→    // `url(#f) blur() saturate()` chain as inert — putting them on one layer
   584→    // makes even the base blur pop in late. Layer A (blur+saturate) never
   585→    // changes after first paint; layer B (displacement only) stacks above it,
   586→    // sampling A's output. The map's neutral center makes B a no-op wherever
   587→    // there's no displacement, so B arriving late just fades refraction in
   588→    // over an already-frosted surface.
   589→    const backdropFilter =
   590→      tier === 'clear'
   591→        ? undefined
   592→        : tier === 'frost'
   593→          ? `blur(${Math.max(effectiveBlur * 2, 10)}px) saturate(${saturation})`
   594→          : `blur(${effectiveBlur}px) saturate(${saturation})`;
   595→    const refractFilter = glassRefs ? `url(#${glassRefs.filterId})` : undefined;
   596→    const tintOpacity = tier === 'clear' ? 1 : 0.25 + 0.75 * frost;
   597→
   598→    return (
   599→      <div
   600→        {...rest}
   601→        ref={localRef}
   602→        className={[
   603→          'liqui-glass',
   604→          `liqui-glass--${tier}`,
   605→          elevated ? 'liqui-glass--elevated' : '',
   606→          className ?? '',
   607→        ]
   608→          .join(' ')
   609→          .trim()}
   610→        style={{ ...style, ['--lq-radius' as string]: `${radius}px` }}
   611→      >
   612→        {backdropFilter && (
   613→          <span
   614→            className="liqui-glass__backdrop"
   615→            style={{ backdropFilter, WebkitBackdropFilter: backdropFilter }}
   616→          />
   617→        )}
   618→        {refractFilter && (
   619→          <span
   620→            className={
   621→              needsFade
   622→                ? 'liqui-glass__refract liqui-glass__refract--fade'
   623→                : 'liqui-glass__refract'
   624→            }
   625→            style={{ backdropFilter: refractFilter, WebkitBackdropFilter: refractFilter }}
   626→          />
   627→        )}
   628→        <span className="liqui-glass__tint" style={{ opacity: tintOpacity }} />
   629→        {glassRefs && specular > 0 && (
   630→          <>
   631→            {/* One image per appearance; CSS shows the matching one, so a
   632→                light/dark switch changes the lighting without JS. */}
   633→            <span
   634→              className={[
   635→                'liqui-glass__specular liqui-glass__specular--light',
   636→                needsFade ? 'liqui-glass__specular--fade' : '',
   637→              ]
   638→                .join(' ')
   639→                .trim()}
   640→              style={{
   641→                backgroundImage: `url(${glassRefs.images.specular})`,
   642→                opacity: specular,
   643→              }}
   644→            />
   645→            <span
   646→              className={[
   647→                'liqui-glass__specular liqui-glass__specular--dark',
   648→                needsFade ? 'liqui-glass__specular--fade' : '',
   649→              ]
   650→                .join(' ')
   651→                .trim()}
   652→              style={{
   653→                backgroundImage: `url(${glassRefs.images.specularDark})`,
   654→                opacity: specular,
   655→              }}
   656→            />
   657→          </>
   658→        )}
   659→        <span className="liqui-glass__shine" />
   660→        <div className={['liqui-glass__content', contentClassName ?? ''].join(' ').trim()}>
   661→          {children}
   662→        </div>
   663→      </div>
   664→    );
   665→  },
   666→);
