    27→// error recovery tuning (see the audio "error" handler)
    28→const RECOVERY_HOLD_MS = 3000;
    29→const WAITING_GRACE_MS = 300; // a buffering hiccup shorter than this never shows // must play this far past a recovery before it may retry again

    31→// A stuck load must never sit in "loading" forever with no feedback (the
    32→// "spinner spins but no error ever appears" failure mode). Both stream
    33→// resolution and the audio.play() call get a deadline; when one fires we
    34→// abort the load and surface a real error (toast + auto-skip + diagnostics)
    35→// instead of hanging silently.
    36→const RESOLVE_DEADLINE_MS = 45_000;
    37→const PLAY_START_DEADLINE_MS = 20_000;

    39→export class PlaybackTimeoutError extends Error {
    40→  constructor(phase: "resolve" | "start", trackTitle: string, ms: number) {
    41→    super(
    42→      phase === "resolve"
    43→        ? `Timed out resolving audio for "${trackTitle}" after ${Math.round(ms / 1000)}s.`
    44→        : `Audio never started for "${trackTitle}" within ${Math.round(ms / 1000)}s.`,
    45→    );
    46→    this.name = "PlaybackTimeoutError";
    47→  }

   167→  private setupMediaSession() {
   168→    if (typeof navigator === "undefined" || !("mediaSession" in navigator))
   169→      return;
   170→
   171→    try {
   172→      navigator.mediaSession.setActionHandler("play", () => {
   173→        usePlayerStore.getState().dispatch({ type: "RESUME_REQUESTED" });
   174→      });
   175→      navigator.mediaSession.setActionHandler("pause", () => {
   176→        usePlayerStore.getState().dispatch({ type: "PAUSE_REQUESTED" });
   177→      });
   178→      navigator.mediaSession.setActionHandler("previoustrack", () => {
   179→        usePlayerStore.getState().dispatch({ type: "SKIP_PREVIOUS" });
   180→      });
   181→      navigator.mediaSession.setActionHandler("nexttrack", () => {
   182→        usePlayerStore.getState().dispatch({ type: "SKIP_NEXT" });
   183→      });
   184→      navigator.mediaSession.setActionHandler("stop", () => {
   185→        usePlayerStore.getState().dispatch({ type: "PAUSE_REQUESTED" });
   186→      });
   187→      const seekBy = (deltaSeconds: number) => {
   188→        const st = usePlayerStore.getState();
   189→        const max =
   190→          st.durationMs > 0 ? st.durationMs : Number.POSITIVE_INFINITY;
   191→        const next = Math.min(
   192→          max,
   193→          Math.max(0, st.positionMs + deltaSeconds * 1000),
   194→        );
   195→        st.dispatch({ type: "SEEK_COMMITTED", positionMs: Math.round(next) });
   196→      };
   197→      navigator.mediaSession.setActionHandler("seekbackward", (details) => {
   198→        seekBy(-(details.seekOffset ?? 10));
   199→      });
   200→      navigator.mediaSession.setActionHandler("seekforward", (details) => {
   201→        seekBy(details.seekOffset ?? 10);
   202→      });
   203→      navigator.mediaSession.setActionHandler("seekto", (details) => {
   204→        if (details.seekTime !== undefined && !Number.isNaN(details.seekTime)) {
   205→          usePlayerStore.getState().dispatch({
   206→            type: "SEEK_COMMITTED",
   207→            positionMs: Math.round(details.seekTime * 1000),
   208→          });
   209→        }
   210→      });
   211→    } catch {
   212→      // ignore unsupported media session actions
   213→    }
   214→  }

   216→  // windows' media controls (and apps built on them, like FluentFlyout) read this.
   217→  // "loading" between tracks still reports playing so the session never looks finished.
   218→  private syncMediaSessionState(status: string) {
   219→    if (typeof navigator === "undefined" || !("mediaSession" in navigator))
   220→      return;
   221→    try {
   222→      navigator.mediaSession.playbackState =
   223→        status === "playing" || status === "loading"
   224→          ? "playing"
   225→          : status === "paused"
   226→            ? "paused"
   227→            : "none";
   228→      if (status === "idle") navigator.mediaSession.metadata = null;
   229→    } catch {
   230→      // ignore unsupported environments
   231→    }
   232→  }

   239→      setPositionState?: (state?: MediaPositionState) => void;

   246→      ms.setPositionState({
   247→        duration,
   248→        playbackRate: this.audio.playbackRate > 0 ? this.audio.playbackRate : 1,
   249→        position: Math.min(duration, Math.max(0, this.audio.currentTime || 0)),
   250→      });
   251→    } catch {
   252→      // ignore invalid transient states
   253→    }

   282→  private setupAudioListeners(audioEl: HTMLAudioElement) {
   283→    audioEl.addEventListener("timeupdate", () => {
   284→      if (this.isResetting || audioEl !== this.audio) return;
   285→      if (!this.currentTrackId) return;
   286→      if (usePlayerStore.getState().status === "loading") return;
   287→      if (Number.isNaN(audioEl.currentTime)) return;
   288→
   289→      const posMs = Math.round(audioEl.currentTime * 1000);
   290→      usePlayerStore.getState().setPosition(posMs);
   291→      if (
   292→        this.recoveryHoldUntilMs !== null &&
   293→        posMs >= this.recoveryHoldUntilMs
   294→      ) {
   295→        this.recoveryHoldUntilMs = null;
   296→        this.recoveryAttempts = 0;
   297→      }
   298→    });
   299→
   300→    audioEl.addEventListener("durationchange", () => {
   301→      if (this.isResetting || audioEl !== this.audio) return;
   302→      if (!this.currentTrackId) return;
   303→      if (Number.isNaN(audioEl.duration) || audioEl.duration <= 0) return;
   304→
   305→      usePlayerStore
   306→        .getState()
   307→        .setDuration(Math.round(audioEl.duration * 1000));
   308→      this.syncMediaSessionPosition();
   309→    });
   310→
   311→    for (const type of [
   312→      "seeked",
   313→      "ratechange",
   314→      "play",
   315→      "pause",
   316→      "loadedmetadata",
   317→    ]) {
   318→      audioEl.addEventListener(type, () => {
   319→        if (this.isResetting || audioEl !== this.audio) return;
   320→        this.syncMediaSessionPosition();
   321→      });
   322→    }
   323→
   324→    audioEl.addEventListener("waiting", () => {
   325→      if (this.isResetting || audioEl !== this.audio) return;
   326→      log("yellow", "audio", "buffering (waiting)");
   327→      this.stallStartTime = performance.now();
   328→      if (this.waitingTimer) clearTimeout(this.waitingTimer);
   329→      this.waitingTimer = setTimeout(() => {
   330→        this.waitingTimer = null;
   331→        // still stuck: a real stall, show it
   332→        if (audioEl !== this.audio || audioEl.paused || audioEl.readyState >= 3)
   333→          return;
   334→        if (usePlayerStore.getState().status === "playing") {
   335→          usePlayerStore.getState().setStatus("loading");
   336→        }
   337→      }, WAITING_GRACE_MS);
   338→    });
   339→
   340→    // resumed (or never really stalled): drop a pending "loading"
   341→    for (const type of ["playing", "seeked", "canplay", "pause", "emptied"]) {
   342→      audioEl.addEventListener(type, () => {
   343→        if (audioEl !== this.audio || !this.waitingTimer) return;
   344→        if (type === "seeked" && audioEl.readyState < 3) return; // seeked but still no data
   345→        clearTimeout(this.waitingTimer);
   346→        this.waitingTimer = null;
   347→      });
   348→    }
   349→
   350→    audioEl.addEventListener("ended", () => {
   351→      if (this.isResetting || audioEl !== this.audio) return;
   352→      if (!this.currentTrackId) return;
   353→      if (this.attachedTrackId !== this.currentTrackId) return;
   354→
   355→      // Ensure the track truly finished playing (not an empty media load or teardown artefact)
   356→      const duration = audioEl.duration;
   357→      const currentTime = audioEl.currentTime;
   358→      if (duration > 0 && currentTime >= Math.max(0, duration - 1.5)) {
   359→        log("cyan", "audio", "ended — skipping to next");
   360→        this.onEnded?.();
   361→        const state = usePlayerStore.getState();
   362→        state.dispatch({ type: "SKIP_NEXT", isAutoEnd: true });
   363→      }
   364→    });
   365→
   366→    audioEl.addEventListener("playing", () => {
   367→      if (this.isResetting || audioEl !== this.audio) return;
   368→      log("green", "audio", "playing");
   369→      if (this.recoveryHoldUntilMs === null) this.recoveryAttempts = 0;
   370→      this.applyPlaybackRateAndPitch();
   371→
   372→      if (this.stallStartTime !== null) {
   373→        const stallDurationMs = Math.round(
   374→          performance.now() - this.stallStartTime,
   375→        );
   376→        if (stallDurationMs > 400) {
   377→          telemetry.trackPlaybackHealth("STALL_BUFFERING", {
   378→            trackId: this.currentTrackId || undefined,
   379→            stallDurationMs,
   380→          });
   381→        }
   382→        this.stallStartTime = null;
   383→      }
   384→
   385→      if (!this.currentTrackId || audioEl.paused) return;
   386→      const currentStatus = usePlayerStore.getState().status;
   387→      if (currentStatus !== "paused") {
   388→        usePlayerStore.getState().setStatus("playing");
   389→      }
   390→    });
   391→
   392→    audioEl.addEventListener("pause", () => {
   393→      if (this.isResetting || audioEl !== this.audio) return;
   394→      log("cyan", "audio", "paused");
   395→      const st = usePlayerStore.getState().status;
   396→      if (st === "playing") {
   397→        usePlayerStore.getState().setStatus("paused");
   398→      }
   399→    });
   400→
   401→    audioEl.addEventListener("error", () => {
   402→      if (this.isResetting || audioEl !== this.audio) return;
   403→      if (
   404→        !this.currentTrackId ||
   405→        !audioEl.src ||
   406→        audioEl.src === window.location.href
   407→      )
   408→        return;
   409→      // the previous track's source is still attached while the next one loads
   410→      if (this.attachedTrackId !== this.currentTrackId) return;
   411→
   412→      const message = audioEl.error?.message || "Audio playback error";
   413→      const code = audioEl.error?.code;
   414→      console.error(
   415→        "[AudioElementError]",
   416→        this.currentTrackId,
   417→        code,
   418→        message,
   419→        audioEl.src,
   420→      );
   421→      log("red", "audio", `error (code ${code}): ${message}`);
   422→
   423→      const isDecode =
   424→        code ===
   425→          (typeof MediaError !== "undefined"
   426→            ? MediaError.MEDIA_ERR_DECODE
   427→            : 3) ||
   428→        code ===
   429→          (typeof MediaError !== "undefined"
   430→            ? MediaError.MEDIA_ERR_SRC_NOT_SUPPORTED
   431→            : 4);
   432→
   433→      if (isDecode) {
   434→        telemetry.trackPlaybackHealth("MEDIA_DECODE_ERROR", {
   435→          trackId: this.currentTrackId || undefined,
   436→          errorMessage: message,
   437→          details: { mediaErrorCode: code },
   438→        });
   439→      }
   440→
   441→      if (this.activeBlobUrl) {
   442→        URL.revokeObjectURL(this.activeBlobUrl);
   443→        this.activeBlobUrl = null;
   444→        void linerDb.deleteAudio(this.currentTrackId);
   445→        audioCache.deleteAudio(this.currentTrackId);
   446→      }
   447→
   448→      const store = usePlayerStore.getState();
   449→      const currentTrack = store.currentTrack;
   450→      const posMs =
   451→        Number.isFinite(audioEl.currentTime) && audioEl.currentTime > 0
   452→          ? Math.round(audioEl.currentTime * 1000)
   453→          : store.positionMs;
   454→
   455→      if (
   456→        currentTrack &&
   457→        currentTrack.id === this.currentTrackId &&
   458→        this.recoveryAttempts < 1
   459→      ) {
   460→        this.recoveryAttempts++;
   461→        this.recoveryHoldUntilMs = posMs + RECOVERY_HOLD_MS;
   462→        log(
   463→          "yellow",
   464→          "playback",
   465→          `audio error encountered — recovering via cache/stream at ${posMs}ms`,
   466→        );
   467→        void this.loadAndPlay(currentTrack, posMs, true);
   468→        return;
   469→      }
   470→
   471→      usePlayerStore.getState().setStatus("error", message);
   472→      this.onError?.({ trackId: this.currentTrackId, message });
   473→    });
   474→  }

   480→  public hasLoadedSourceFor(trackId: string): boolean {
   481→    const isExpired =
   482→      this.currentSessionExpiresAt !== null &&
   483→      Date.now() >= this.currentSessionExpiresAt - 15_000;
   484→    return (
   485→      this.currentTrackId === trackId &&
   486→      this.attachedTrackId === trackId &&
   487→      Boolean(this.audio.currentSrc || this.audio.src) &&
   488→      this.audio.src !== window.location.href &&
   489→      !this.audio.error &&
   490→      !isExpired
   491→    );
   492→  }

   494→  public applyPlaybackRateAndPitch(
   495→    rate?: number,
   496→    preservesPitch?: boolean,
   497→  ): void {
   498→    if (!this.audio) return;
   499→    const store = usePlayerStore.getState();
   500→    const targetRate = rate ?? store.playbackRate ?? 1;
   501→    const targetPreserves =
   502→      preservesPitch ??
   503→      (store.isPitchLinked !== undefined ? !store.isPitchLinked : true);
   504→
   505→    try {
   506→      this.audio.playbackRate = targetRate;
   507→      this.audio.defaultPlaybackRate = targetRate;
   508→      this.audio.preservesPitch = targetPreserves;
   509→      (this.audio as any).mozPreservesPitch = targetPreserves;
   510→      (this.audio as any).webkitPreservesPitch = targetPreserves;
   511→    } catch {
   512→      // Audio element not initialized or properties unsupported
   513→    }
   514→
   515→    try {
   516→      const isPitchLinked = store.isPitchLinked ?? true;
   517→      const pitchSemitones = store.pitchSemitones ?? 0;
   518→      const isReverbEnabled = store.isReverbEnabled ?? false;
   519→      const reverbLevel = store.reverbLevel ?? 0.35;
   520→      const needsEffects =
   521→        isReverbEnabled ||
   522→        (!isPitchLinked && Math.abs(pitchSemitones) >= 0.05) ||
   523→        isEqActive();
   524→      if (!needsEffects && !pitchShifter.isAttached()) return;
   525→      pitchShifter.attach(this.audio);
   526→      pitchShifter.update({
   527→        pitchSemitones,
   528→        isPitchLinked,
   529→        isReverbEnabled,
   530→        reverbLevel,
   531→      });
   532→      // re-assert EQ after (re)attach so a fresh graph serves the saved curve
   533→      ensureEqualizerApplied();
   534→    } catch (err) {
   535→      console.error("[PlayerRuntime] applyPlaybackRateAndPitch error:", err);
   536→    }
   537→  }

   546→  private fadeVolume(
   547→    targetVolume: number,
   548→    durationMs: number = 180,
   549→    onComplete?: () => void,
   550→  ): void {
   551→    this.cancelFade();
   552→    const startVolume = this.audio.volume;
   553→    const diff = targetVolume - startVolume;
   554→    if (Math.abs(diff) < 0.001 || durationMs <= 0) {
   555→      this.audio.volume = Math.max(0, Math.min(1, targetVolume));
   556→      onComplete?.();
   557→      return;
   558→    }
   559→
   560→    const startTime = performance.now();
   561→
   562→    const tick = () => {
   563→      const now = performance.now();
   564→      const elapsed = now - startTime;
   565→      const progress = Math.min(elapsed / durationMs, 1);
   566→      const eased = 0.5 - 0.5 * Math.cos(progress * Math.PI);
   567→      const current = startVolume + diff * eased;
   568→      this.audio.volume = Math.max(0, Math.min(1, current));
   569→
   570→      if (progress < 1) {
   571→        // audio fade ticks independently of display vsync in background
   572→        this.fadeTimer = setTimeout(tick, 16);
   573→      } else {
   574→        this.fadeTimer = null;
   575→        this.audio.volume = Math.max(0, Math.min(1, targetVolume));
   576→        onComplete?.();
   577→      }
   578→    };
   579→
   580→    this.fadeTimer = setTimeout(tick, 16);
   581→  }
   582→
   583→  private releaseStream() {
   584→    this.releaseActiveStream?.();
   585→    this.releaseActiveStream = null;
   586→  }
   587→
   588→  private watchFullSource(trackId: string, stream: PlaybackStream) {
   589→    const { completed, needsFullSource } = stream;
   590→    if (!completed || !needsFullSource) return;
   591→    void completed
   592→      .then((blob) => {
   593→        if (
   594→          this.currentTrackId !== trackId ||
   595→          this.audio.src !== rewriteForWeb(stream.url)
   596→        )
   597→          return;
   598→        if (needsFullSource()) this.swapToFullSource(blob);
   599→      })
   600→      .catch(() => undefined);
   601→  }
   602→
   603→  private swapToFullSource(blob: Blob) {
   604→    const position = this.audio.currentTime;
   605→    const wasPlaying = !this.audio.paused;
   606→    const url = URL.createObjectURL(blob);
   607→    this.isResetting = true;
   608→    this.releaseStream();
   609→    if (this.activeBlobUrl) URL.revokeObjectURL(this.activeBlobUrl);
   610→    this.activeBlobUrl = url;
   611→    this.audio.src = url;
   612→    this.applyPlaybackRateAndPitch();
   613→    const settle = () => {
   614→      this.isResetting = false;
   615→    };
   616→    this.audio.addEventListener(
   617→      "loadedmetadata",
   618→      () => {
   619→        this.audio.currentTime = position;
   620→        if (wasPlaying) void this.audio.play().then(settle, settle);
   621→        else settle();
   622→      },
   623→      { once: true },
   624→    );
   625→    log("cyan", "playback", "switched to the fully downloaded source");
   626→  }
   627→
   628→  private safeResetAudioElement() {
   629→    this.isResetting = true;
   630→    this.currentSessionExpiresAt = null;
   631→    try {
   632→      this.cancelFade();
   633→      this.audio.pause();
   634→      this.releaseStream();
   635→      if (this.activeBlobUrl) {
   636→        URL.revokeObjectURL(this.activeBlobUrl);
   637→        this.activeBlobUrl = null;
   638→      }
   639→      this.audio.removeAttribute("src");
   640→      this.audio.load();
   641→      this.attachedTrackId = null;
   642→    } catch {
   643→    } finally {
   644→      this.isResetting = false;
   645→    }
   646→  }

   648→  // between tracks: silence the old source but keep it attached until the next one
   649→  // is ready. Emptying the element here made Chromium drop the system media session
   650→  // during every skip, so Windows controls / FluentFlyout lost Liner after a track change.
   651→  private pauseForTransition() {
   652→    this.isResetting = true;
   653→    this.currentSessionExpiresAt = null;
   654→    try {
   655→      this.cancelFade();
   656→      this.audio.pause();
   657→    } catch {
   658→    } finally {
   659→      this.isResetting = false;
   660→    }
   661→  }

   663→  private attachSource(url: string, isBlob: boolean, trackId: string) {
   664→    const previousBlob = this.activeBlobUrl;
   665→    // web port: route remote media through the same-origin media proxy so the
   666→    // crossOrigin="anonymous" audio element stays readable for the visualizer
   667→    this.audio.src = rewriteForWeb(url);
   668→    this.activeBlobUrl = isBlob ? url : null;
   669→    this.attachedTrackId = trackId;
   670→    if (previousBlob && previousBlob !== url) URL.revokeObjectURL(previousBlob);
   671→  }

   676→    this.loadAbortController = null;
   677→    this.backgroundCacheAbortController?.abort();
   678→    this.backgroundCacheAbortController = null;
   679→    this.currentTrackId = null;
   680→    this.currentSessionExpiresAt = null;
   681→    this.recoveryAttempts = 0;
   682→    this.recoveryHoldUntilMs = null;
   683→    this.safeResetAudioElement();

   688→  public async preloadTrack(track: Track): Promise<void> {
   689→    if (this.currentTrackId === track.id) return;
   690→    if (track.localFileId) return; // local blobs are already "cached"
   691→    if (this.preloadingTrackId === track.id) return;
   692→
   693→    const cached = await linerDb.getAudio(track.id);
   694→    if (cached) return;
   695→
   696→    this.preloadAbortController?.abort();
   697→    const controller = new AbortController();
   698→    this.preloadAbortController = controller;
   699→    this.preloadingTrackId = track.id;
   700→
   701→    const generation = audioCache.currentGeneration();
   702→    const promise = (async () => {
   703→      try {
   704→        const stream = await resolvePlaybackStream(track, controller.signal);
   705→        if (controller.signal.aborted) return;
   706→
   707→        const blob = await this.downloadAudioBlob(stream, controller.signal);
   708→        if (!blob || controller.signal.aborted) return;
   709→        if (
   710→          !(await audioCache.saveAudioBlob(
   711→            track.id,
   712→            blob,
   713→            blob.type,
   714→            generation,
   715→          ))
   716→        )
   717→          return;
   718→        await linerDb.putTrack(track);
   719→        log(
   720→          "green",
   721→          "preload",
   722→          `preloaded audio: "${track.title}" (${Math.round(blob.size / 1024)} KB)`,
   723→        );
   724→      } catch {
   725→      } finally {
   726→        if (this.preloadingTrackId === track.id) {
   727→          this.preloadingTrackId = null;
   728→          this.preloadingPromise = null;
   729→          this.preloadAbortController = null;
   730→        }
   731→      }
   732→    })();
   733→
   734→    this.preloadingPromise = promise;
   735→    await promise;
   736→  }

   772→  public async loadAndPlay(
   773→    track: Track,
   774→    startPositionMs: number = 0,
   775→    bypassCache: boolean = false,
   776→  ) {
   777→    this.cancelFade();
   778→    this.pendingStartPositionMs = startPositionMs;
   779→    const epoch = ++this.loadEpoch;
   780→
   781→    if (this.currentTrackId !== track.id) {
   782→      this.recoveryAttempts = 0;
   783→      this.recoveryHoldUntilMs = null;
   784→    }
   785→
   786→    if (this.preloadingTrackId === track.id && this.preloadingPromise) {
   787→      await this.preloadingPromise;
   788→    } else if (this.preloadingTrackId && this.preloadingTrackId !== track.id) {
   789→      this.preloadAbortController?.abort();
   790→      this.preloadAbortController = null;
   791→      this.preloadingTrackId = null;
   792→      this.preloadingPromise = null;
   793→    }
   794→
   795→    this.loadAbortController?.abort();
   796→    const controller = new AbortController();
   797→    this.loadAbortController = controller;
   798→
   799→    this.backgroundCacheAbortController?.abort();
   800→    const bgCacheController = new AbortController();
   801→    this.backgroundCacheAbortController = bgCacheController;
   802→
   803→    const store = usePlayerStore.getState();
   804→    if (!store.keepSpeedAcrossTracks && this.currentTrackId !== track.id) {
   805→      store.resetSpeedAndPitch();
   806→    }
   807→    this.currentTrackId = track.id;
   808→    this.loadStartTime = performance.now();
   809→    this.updateMediaSessionMetadata(track);
   810→
   811→    store.setStatus("loading", null);
   812→    store.setPosition(startPositionMs > 0 ? startPositionMs : 0);
   813→
   814→    this.pauseForTransition();
   815→
   816→    try {
   817→      const cachedAudio = bypassCache ? null : await linerDb.getAudio(track.id);
   818→      if (epoch !== this.loadEpoch || controller.signal.aborted) {
   819→        return;
   820→      }
   821→
   822→      const isValidCachedBlob =
   823→        Boolean(cachedAudio?.blob) &&
   824→        cachedAudio!.blob.size > 0 &&
   825→        (!cachedAudio!.blob.type ||
   826→          cachedAudio!.blob.type.startsWith("audio/"));
   827→
   828→      if (cachedAudio && !isValidCachedBlob) {
   829→        void linerDb.deleteAudio(track.id);
   830→        audioCache.deleteAudio(track.id);
   831→      }
   832→
   833→      if (cachedAudio && isValidCachedBlob) {
   834→        this.currentSessionExpiresAt = null;
   835→        void linerDb.putTrack(track);
   836→        const blobUrl = URL.createObjectURL(cachedAudio.blob);
   837→        this.attachSource(blobUrl, true, track.id);
   838→        this.audio.volume = 0;
   839→        this.audio.load();
   840→        this.applyPlaybackRateAndPitch();
   841→
   842→        const targetPosition =
   843→          this.pendingStartPositionMs > 0
   844→            ? this.pendingStartPositionMs
   845→            : startPositionMs;
   846→        this.seekWhenReady(epoch, targetPosition);
   847→        log(
   848→          "cyan",
   849→          "playback",
   850→          `audio cache hit for "${track.title}" (${Math.round(cachedAudio.byteSize / 1024)} KB)`,
   851→        );
   852→
   853→        if (this.audio.readyState < 3) {
   854→          await new Promise<void>((resolve) => {
   855→            const timeout = setTimeout(resolve, 3000);
   856→            const onCanPlay = () => {
   857→              clearTimeout(timeout);
   858→              this.audio.removeEventListener("canplay", onCanPlay);
   859→              this.audio.removeEventListener("error", onCanPlay);
   860→              resolve();
   861→            };
   862→            this.audio.addEventListener("canplay", onCanPlay, { once: true });
   863→            this.audio.addEventListener("error", onCanPlay, { once: true });
   864→          });
   865→        }
   866→
   867→        if (epoch !== this.loadEpoch || controller.signal.aborted) return;
   868→
   869→        const playPromise = this.audio.play();
   870→        this.activePlayPromise = playPromise;
   871→        await withDeadline(
   872→          playPromise,
   873→          PLAY_START_DEADLINE_MS,
   874→          () =>
   875→            new PlaybackTimeoutError(
   876→              "start",
   877→              track.title,
   878→              PLAY_START_DEADLINE_MS,
   879→            ),
   880→        );
   881→
   882→        const audioStartLatencyMs = Math.round(
   883→          performance.now() - this.loadStartTime,
   884→        );
   885→
   886→        if (epoch === this.loadEpoch) {
   887→          usePlayerStore.getState().setStatus("playing");
   888→          this.fadeVolume(usePlayerStore.getState().volume, 180);
   889→        }
   890→        log(
   891→          "green",
   892→          "playback",
   893→          `started (cached): ${track.title} in ${audioStartLatencyMs}ms`,
   894→        );
   895→        return;
   896→      }
   897→
   898→      const stream = await withDeadline(
   899→        resolvePlaybackStream(track, controller.signal, {
   900→          progressive: true,
   901→        }),
   902→        RESOLVE_DEADLINE_MS,
   903→        () =>
   904→          new PlaybackTimeoutError("resolve", track.title, RESOLVE_DEADLINE_MS),
   905→      );
   906→      this.ttfbMs = Math.round(performance.now() - this.loadStartTime);
   907→
   908→      if (epoch !== this.loadEpoch || controller.signal.aborted) {
   909→        stream.release?.();
   910→        log(
   911→          "yellow",
   912→          "playback",
   913→          `superseded before stream ready (${track.title})`,
   914→        );
   915→        return;
   916→      }
   917→
   918→      const streamCodec = stream.codec;
   919→      const streamBitrate = stream.bitrate;
   920→      this.currentSessionExpiresAt = stream.expiresAtMs;
   921→      log("cyan", "playback", `resolved stream for ${track.title}`);
   922→      this.releaseStream();
   923→      const url = stream.blob ? URL.createObjectURL(stream.blob) : stream.url;
   924→      this.attachSource(url, Boolean(stream.blob), track.id);
   925→      this.releaseActiveStream = stream.release ?? null;
   926→      this.watchFullSource(track.id, stream);
   927→      this.audio.volume = 0;
   928→      this.audio.load();
   929→      this.applyPlaybackRateAndPitch();
   930→
   931→      const targetPosition =
   932→        this.pendingStartPositionMs > 0
   933→          ? this.pendingStartPositionMs
   934→          : startPositionMs;
   935→      this.seekWhenReady(epoch, targetPosition);
   936→      log(
   937→        "green",
   938→        "playback",
   939→        `stream attached: ${streamCodec} ${streamBitrate} bps (${track.title})`,
   940→      );
   941→
   942→      if (this.audio.readyState < 3) {
   943→        await new Promise<void>((resolve) => {
   944→          const timeout = setTimeout(resolve, 3000);
   945→          const onCanPlay = () => {
   946→            clearTimeout(timeout);
   947→            this.audio.removeEventListener("canplay", onCanPlay);
   948→            this.audio.removeEventListener("error", onCanPlay);
   949→            resolve();
   950→          };
   951→          this.audio.addEventListener("canplay", onCanPlay, { once: true });
   952→          this.audio.addEventListener("error", onCanPlay, { once: true });
   953→        });
   954→      }
   955→
   956→      if (epoch !== this.loadEpoch || controller.signal.aborted) return;
   957→
   958→      const playPromise = this.audio.play();
   959→      this.activePlayPromise = playPromise;
   960→      await withDeadline(
   961→        playPromise,
   962→        PLAY_START_DEADLINE_MS,
   963→        () =>
   964→          new PlaybackTimeoutError("start", track.title, PLAY_START_DEADLINE_MS),
   965→      );
   966→
   967→      const audioStartLatencyMs = Math.round(
   968→        performance.now() - this.loadStartTime,
   969→      );
   970→
   971→      if (epoch === this.loadEpoch) {
   972→        usePlayerStore.getState().setStatus("playing");
   973→        this.fadeVolume(usePlayerStore.getState().volume, 180);
   974→        this.cacheAudioInBackground(track, stream, bgCacheController.signal);
   975→      }
   976→      log(
   977→        "green",
   978→        "playback",
   979→        `started: ${track.title} in ${audioStartLatencyMs}ms`,
   980→      );
   981→
   982→      telemetry.trackPlaybackHealth("INIT_LATENCY", {
   983→        trackId: track.id,
   984→        codec: streamCodec,
   985→        bitrate: streamBitrate,
   986→        timeToFirstByteMs: this.ttfbMs,
   987→        audioStartLatencyMs,
   988→      });
   989→    } catch (error) {
   990→      if (epoch !== this.loadEpoch || controller.signal.aborted) {
   991→        return;
   992→      }
   993→
   994→      if (error instanceof PlaybackTimeoutError) {
   995→        // deadline fired while something was still pending — stop the
   996→        // underlying work and record it so the Settings diagnostics panel
   997→        // shows WHY this track failed instead of a silent hang
   998→        controller.abort();
   999→        recordPlaybackFailure("unresolved", error, track.title);
  1000→      }
  1001→
  1002→      if (error instanceof DOMException && error.name === "AbortError") {
  1003→        return;
  1004→      }
  1005→
  1006→      if (error instanceof DOMException && error.name === "NotAllowedError") {
  1007→        log(
  1008→          "yellow",
  1009→          "playback",
  1010→          "autoplay blocked — waiting for user gesture",
  1011→        );
  1012→        usePlayerStore.getState().setStatus("paused");
  1013→        return;
  1014→      }
  1015→
  1016→      if (
  1017→        isBotCheckError(error) &&
  1018→        !isYoutubeSignedIn() &&
  1019→        !isYoutubeSignInDeclined() &&
  1020→        prefersSignInPrompt()
  1021→      ) {
  1022→        log("yellow", "playback", "youtube sign-in required, pausing");
  1023→        usePlayerStore.getState().setStatus("paused");
  1024→        requestYoutubeSignIn();
  1025→        return;
  1026→      }
  1027→
  1028→      const message =
  1029→        error instanceof ApiError
  1030→          ? error.message
  1031→          : error instanceof Error
  1032→            ? error.message
  1033→            : "Playback failed to start.";
  1034→      console.error("[LoadAndPlayError]", track.id, error);
  1035→      log("red", "playback", `failed: ${message}`);
  1036→      usePlayerStore.getState().setStatus("error", message);
  1037→
  1038→      telemetry
  1039→        .reportError(error, {
  1040→          action: "playback_failed",
  1041→          trackId: track.id,
  1042→          title: track.title,
  1043→        })
  1044→        .catch(() => {});
  1045→
  1046→      this.onError?.({ trackId: track.id, message });
  1047→    } finally {
  1048→      if (epoch === this.loadEpoch) {
  1049→        this.activePlayPromise = null;
  1050→      }
  1051→    }
  1052→  }

  1054→  private seekWhenReady(epoch: number, positionMs: number) {
  1055→    if (positionMs <= 0) return;
  1056→    const apply = () => {
  1057→      if (epoch !== this.loadEpoch) return;
  1058→      try {
  1059→        this.audio.currentTime = positionMs / 1000;
  1060→      } catch {
  1061→        // Media not seekable yet — ignore
  1062→      }
  1063→    };
  1064→    if (this.audio.readyState >= 1) {
  1065→      apply();
  1066→    } else {
  1067→      this.audio.addEventListener("loadedmetadata", apply, { once: true });
  1068→    }
  1069→  }

  1074→    this.loadAbortController = null;
  1075→    usePlayerStore.getState().setStatus("paused");
  1076→    log("cyan", "audio", "pause requested (fading out)");
  1077→

  1082→  }

  1084→  public resume() {
  1085→    const epoch = ++this.loadEpoch;
  1086→    this.loadAbortController?.abort();
  1087→    this.loadAbortController = null;
  1088→
  1089→    const targetVolume = usePlayerStore.getState().volume;
  1090→    usePlayerStore.getState().setStatus("playing");
  1091→    pitchShifter.resumeContext();
  1092→    this.applyPlaybackRateAndPitch();
  1093→    log("cyan", "audio", "resume requested (fading in)");
  1094→
  1095→    const currentTrack = usePlayerStore.getState().currentTrack;
  1096→    const isExpired =
  1097→      this.currentSessionExpiresAt !== null &&
  1098→      Date.now() >= this.currentSessionExpiresAt - 15_000;
  1099→    const hasMediaError = Boolean(this.audio.error);
  1100→    const hasValidSource =
  1101→      Boolean(this.audio.src) &&
  1102→      this.audio.src !== window.location.href &&
  1103→      !isExpired &&
  1104→      !hasMediaError;
  1105→
  1106→    if (!hasValidSource && currentTrack) {
  1107→      void this.loadAndPlay(currentTrack, usePlayerStore.getState().positionMs);
  1108→      return;
  1109→    }
  1110→
  1111→    if (this.audio.paused) {
  1112→      this.audio.volume = 0;
  1113→      const playPromise = this.audio.play();
  1114→      this.activePlayPromise = playPromise;
  1115→      playPromise
  1116→        .then(() => {
  1117→          if (epoch !== this.loadEpoch) return;
  1118→          this.fadeVolume(targetVolume, 180);
  1119→        })
  1120→        .catch((error) => {
  1121→          if (epoch !== this.loadEpoch) return;
  1122→          if (error.name === "NotAllowedError") {
  1123→            log("yellow", "audio", "play blocked — waiting for user gesture");
  1124→            usePlayerStore.getState().setStatus("paused");
  1125→          } else if (error.name !== "AbortError") {
  1126→            log("red", "audio", `play failed: ${String(error)}`);
  1127→            if (currentTrack && this.recoveryAttempts < 1) {
  1128→              this.recoveryAttempts++;
  1129→              log(
  1130→                "yellow",
  1131→                "playback",
  1132→                "resume failed — attempting cache/stream recovery",
  1133→              );
  1134→              void this.loadAndPlay(
  1135→                currentTrack,
  1136→                usePlayerStore.getState().positionMs,
  1137→                true,
  1138→              );
  1139→              return;
  1140→            }
  1141→            usePlayerStore.getState().setStatus("paused");
  1142→          }
  1143→        })
  1144→        .finally(() => {
  1145→          if (epoch === this.loadEpoch) this.activePlayPromise = null;
  1146→        });
  1147→    } else {
  1148→      this.fadeVolume(targetVolume, 180);
  1149→    }
  1150→  }

  1162→              this.audio.currentTime = positionMs / 1000;
  1163→            } catch {}
  1164→          },
  1165→          { once: true },

  1167→      }

  1173→  private async downloadAudioBlob(
  1174→    stream: PlaybackStream,
  1175→    signal: AbortSignal,
  1176→  ): Promise<Blob | null> {
  1177→    if (stream.blob) return stream.blob;
  1178→    if (stream.completed) return stream.completed;
  1179→    const startedAt = performance.now();
  1180→    const bytes = await downloadAudioBytes(stream.url, {
  1181→      contentLength: stream.contentLength,
  1182→      signal,
  1183→    });
  1184→    if (!bytes || signal.aborted) return null;
  1185→    const mimeType = stream.mimeType || "audio/webm";
  1186→    if (!mimeType.startsWith("audio/") && !mimeType.startsWith("video/"))
  1187→      return null;
  1188→    if (!isStandaloneAudioFile(bytes.subarray(0, 16))) {
  1189→      log(
  1190→        "yellow",
  1191→        "download",
  1192→        `not caching: stream is not a standalone audio file`,
  1193→      );
  1194→      return null;
  1195→    }
  1196→    log(
  1197→      "cyan",
  1198→      "download",
  1199→      `${Math.round(bytes.byteLength / 1024)} KB in ${Math.round(performance.now() - startedAt)}ms (${stream.contentLength ? "chunked" : "single request"})`,
  1200→    );
  1201→    return new Blob([bytes], { type: mimeType });
  1202→  }

  1204→  private cacheAudioInBackground(
  1205→    track: Track,
  1206→    stream: PlaybackStream,
  1207→    signal: AbortSignal,
  1208→  ) {
  1209→    if (track.localFileId) return; // never duplicate the user's own files
  1210→    if (track.durationMs && track.durationMs > 20 * 60 * 1000) return;
  1211→    const generation = audioCache.currentGeneration();
  1212→    void (async () => {
  1213→      try {
  1214→        const blob = await this.downloadAudioBlob(stream, signal);
  1215→        if (!blob || signal.aborted) return;
  1216→        if (
  1217→          !(await audioCache.saveAudioBlob(
  1218→            track.id,
  1219→            blob,
  1220→            blob.type,
  1221→            generation,
  1222→          ))
  1223→        )
  1224→          return;
  1225→        await linerDb.putTrack(track);
  1226→        log(
  1227→          "green",
  1228→          "playback",
  1229→          `background cached: ${track.title} (${Math.round(blob.size / 1024)} KB)`,
  1230→        );
  1231→      } catch {}
  1232→    })();
  1233→  }
