   343→export default function SettingsModal() {
   344→  const { t } = useTranslation();
   345→  const open = useModalStore((state) => state.settingsOpen);
   346→  const close = useModalStore((state) => state.closeSettings);
   347→
   348→  const [activeTab, setActiveTab] = useState<TabId>("Account");
   349→  const [searchQuery, setSearchQuery] = useState("");
   350→  const [searchFocused, setSearchFocused] = useState(false);
   351→  const {
   352→    attach: attachNavFade,
   353→    onScroll: onNavScroll,
   354→    style: navFadeStyle,
   355→  } = useScrollFade<HTMLElement>();
   356→  const searchCentered = !searchQuery && !searchFocused;
   357→
   358→  const handleClose = () => {
   359→    setSearchQuery("");
   360→    close();
   361→  };
   362→
   363→  const tabLabels: Record<TabId, string> = {
   364→    Account: t("settings.tabs.account") || "Account",
   365→    Playback: t("settings.tabs.playback"),
   366→    Shortcuts: t("settings.tabs.shortcuts") || "Shortcuts",
   367→    Overlay: t("settings.tabs.overlay") || "Overlay",
   368→    Discord: t("settings.tabs.discord") || "Discord RPC",
   369→    Youtube: t("settings.tabs.youtube") || "YouTube",
   370→    Appearance: t("settings.tabs.appearance"),
   371→    Customization: t("settings.tabs.customization") || "Customization",
   372→    Storage: t("settings.tabs.storage"),
   373→    About: t("settings.tabs.about"),
   374→  };
   375→
   376→  const tabDescriptions: Record<TabId, string> = {
   377→    Account:
   378→      t("settings.profile.description") ||
   379→      "Update your public profile, avatar and privacy preferences.",
   380→    Playback: t("settings.playback.description"),
   381→    Shortcuts:
   382→      t("settings.shortcuts.description") ||
   383→      "System-wide and player keyboard shortcuts.",
   384→    Overlay:
   385→      t("settings.overlay.description") ||
   386→      "Desktop overlay behavior, position and lyrics display.",
   387→    Discord:
   388→      t("settings.discord.description") ||
   389→      "Show what you're listening to on Discord with artwork, artist badge and progress bar.",
   390→    Youtube:
   391→      t("settings.youtube.description") ||
   392→      "Sign in to your YouTube account to prevent playback errors.",
   393→    Appearance: t("settings.theme.description"),
   394→    Customization:
   395→      t("settings.customization.description") ||
   396→      "Window background wallpaper, transparency and block blur effects.",
   397→    Storage: t("settings.storage.description"),
   398→    About: t("settings.about.tagline"),
   399→  };
   400→
   401→  const matchingTabIds = useMemo(() => {
   402→    if (!searchQuery.trim()) return null;
   403→    const query = searchQuery.toLowerCase().trim();
   404→    const set = new Set<TabId>();
   405→
   406→    const TAB_LABEL_KEYS: Record<TabId, string> = {
   407→      Account: "settings.tabs.account",
   408→      Playback: "settings.tabs.playback",
   409→      Shortcuts: "settings.tabs.shortcuts",
   410→      Overlay: "settings.tabs.overlay",
   411→      Discord: "settings.tabs.discord",
   412→      Youtube: "settings.tabs.youtube",
   413→      Appearance: "settings.theme.title",
   414→      Customization: "settings.tabs.customization",
   415→      Storage: "settings.tabs.storage",
   416→      About: "settings.tabs.about",
   417→    };
   418→
   419→    const TAB_DESC_KEYS: Record<TabId, string> = {
   420→      Account: "settings.profile.description",
   421→      Playback: "settings.playback.description",
   422→      Shortcuts: "settings.shortcuts.description",
   423→      Overlay: "settings.overlay.description",
   424→      Discord: "settings.discord.description",
   425→      Youtube: "settings.youtube.description",
   426→      Appearance: "settings.theme.description",
   427→      Customization: "settings.customization.description",
   428→      Storage: "settings.storage.description",
   429→      About: "settings.about.tagline",
   430→    };
   431→
   432→    (Object.keys(TAB_LABEL_KEYS) as TabId[]).forEach((tabId) => {
   433→      const texts = [
   434→        ...getTranslationsForAllLocales(TAB_LABEL_KEYS[tabId]),
   435→        ...getTranslationsForAllLocales(TAB_DESC_KEYS[tabId]),
   436→      ];
   437→      if (texts.some((txt) => txt.toLowerCase().includes(query))) {
   438→        set.add(tabId);
   439→      }
   440→    });
   441→
   442→    SETTING_ITEMS.forEach((item) => {
   443→      const texts = [
   444→        ...getTranslationsForAllLocales(item.titleKey),
   445→        ...(item.descKey ? getTranslationsForAllLocales(item.descKey) : []),
   446→      ];
   447→      if (texts.some((txt) => txt.toLowerCase().includes(query))) {
   448→        set.add(item.tabId);
   449→      }
   450→    });
   451→
   452→    return set;
   453→  }, [searchQuery]);
   454→
   455→  const filteredGroups = useMemo(() => {
   456→    if (!matchingTabIds) return TAB_GROUPS;
   457→

   459→      const ids = group.ids.filter((id) => matchingTabIds.has(id));
   460→      return { ...group, ids };

   462→  }, [matchingTabIds]);
   463→

   465→    if (!searchQuery.trim()) return;
   466→    if (
   467→      matchingTabIds &&
   468→      matchingTabIds.size > 0 &&
   469→      !matchingTabIds.has(activeTab)
   470→    ) {
   471→      const firstMatch = TAB_GROUPS.flatMap((g) => g.ids).find((id) =>
   472→        matchingTabIds.has(id),
   473→      );
   474→      if (firstMatch) {
   475→        setActiveTab(firstMatch);
   476→      }
   477→    }
   478→  }, [searchQuery, matchingTabIds, activeTab]);
   479→
   480→  const openExternal = (url: string) => {
   481→    window.open(url, "_blank", "noopener,noreferrer");
   482→  };
   483→
   484→  const user = useAuthStore((state) => state.user);
   485→  const logout = useAuthStore((state) => state.logout);
   486→  const displayName = userDisplayName(user) ?? "Liner";
   487→  const handle = user?.username
   488→    ? `@${user.username}`
   489→    : (user?.email ?? t("settings.manage_account"));
   490→
   491→  return (
   492→    <Dialog
   493→      open={open}
   494→      onOpenChange={(next) => (!next ? handleClose() : undefined)}
   495→      maxWidth={900}
   496→      className="p-0 h-[700px] !max-h-[90vh] !flex-row overflow-hidden !bg-bg-surface/70 dark:!bg-bg-elevated/50 !backdrop-blur-[60px] !border-0"
   497→    >
   498→      <aside className="w-[248px] shrink-0 flex flex-col min-h-0 bg-transparent">
   499→        <div className="px-[12px] pt-[14px] pb-[8px]">
   500→          <div className="relative flex items-center">
   501→            <span
   502→              aria-hidden
   503→              className={`pointer-events-none absolute inset-y-0 flex items-center gap-[6px] text-text-tertiary text-[13px] transition-all duration-200 ${
   504→                searchCentered ? "left-1/2 -translate-x-1/2" : "left-[10px]"
   505→              }`}
   506→              style={font}
   507→            >
   508→              <Search2Line size={15} className="shrink-0" />
   509→              {!searchQuery && (
   510→                <span className="whitespace-nowrap">
   511→                  {t("settings.search_placeholder")}
   512→                </span>
   513→              )}
   514→            </span>
   515→            <input
   516→              type="text"
   517→              value={searchQuery}
   518→              onChange={(e) => setSearchQuery(e.target.value)}
   519→              onFocus={() => setSearchFocused(true)}
   520→              onBlur={() => setSearchFocused(false)}
   521→              aria-label={t("settings.search_placeholder")}
   522→              className="w-full h-[34px] pl-[28px] pr-[26px] bg-bg-surface/70 dark:bg-white/[0.08] text-text-primary text-[13px] rounded-[10px] border-0 outline-none placeholder:text-text-tertiary focus:bg-bg-surface dark:focus:bg-white/[0.12] transition-colors"
   523→              style={font}
   524→            />
   525→            {searchQuery && (
   526→              <button
   527→                type="button"
   528→                onClick={() => setSearchQuery("")}
   529→                className="absolute right-[6px] p-0 border-0 bg-transparent text-text-tertiary hover:text-text-primary cursor-pointer flex items-center justify-center"
   530→              >
   531→                <CloseCircleFill size={14} />
   532→              </button>
   533→            )}
   534→          </div>
   535→        </div>
   536→
   537→        <nav
   538→          ref={attachNavFade}
   539→          onScroll={onNavScroll}
   540→          style={navFadeStyle}
   541→          className="flex-1 min-h-0 overflow-y-auto px-[12px] pb-[12px] flex flex-col gap-[10px]"
   542→          role="tablist"
   543→          aria-label={t("settings.title")}
   544→        >
   545→          {filteredGroups.length === 0 ? (
   546→            <p
   547→              className="text-text-tertiary text-[12px] px-[10px] py-[8px] m-0"
   548→              style={font}
   549→            >
   550→              {t("search.no_results") || "Ничего не найдено"}
   551→            </p>
   552→          ) : (
   553→            filteredGroups.map((group) => (
   554→              <div key={group.labelKey} className="flex flex-col">
   555→                <p
   556→                  className="text-text-tertiary text-[11px] font-[500] m-0 px-[10px] pb-[4px] pt-[2px]"
   557→                  style={font}
   558→                >
   559→                  {t(`settings.groups.${group.labelKey}`)}
   560→                </p>
   561→                <div className={navGroupClass}>
   562→                  {group.ids.map((id) => {
   563→                    const isActive = id === activeTab;
   564→                    const Icon = TAB_ICONS[id];
   565→                    return (
   566→                      <button
   567→                        key={id}
   568→                        type="button"
   569→                        role="tab"
   570→                        aria-selected={isActive}
   571→                        onClick={() => setActiveTab(id)}
   572→                        className={`w-full flex items-center gap-[10px] px-[10px] py-[7px] rounded-[10px] border-0 cursor-pointer text-left transition-colors ${
   573→                          isActive
   574→                            ? "bg-transparent text-text-primary"
   575→                            : "bg-transparent text-text-secondary hover:bg-black/[0.04] hover:text-text-primary dark:hover:bg-white/[0.05]"
   576→                        }`}
   577→                        style={{
   578→                          ...font,
   579→                          fontSize: "13px",
   580→                          fontWeight: isActive ? 600 : 400,
   581→                        }}
   582→                      >
   583→                        <Icon
   584→                          size={18}
   585→                          weight="Bold"
