    50→async function fetchWithTimeout(
    51→  url: string,
    52→  init?: RequestInit,
    53→  timeoutMs = DEFAULT_TIMEOUT_MS,
    54→): Promise<Response> {
    55→  const controller = new AbortController();
    56→  const userSignal = init?.signal;
    57→  let didTimeout = false;
    58→
    59→  const timer = setTimeout(() => {
    60→    didTimeout = true;
    61→    controller.abort(new Error(`Request timed out after ${timeoutMs}ms`));
    62→  }, timeoutMs);
    63→
    64→  let onUserAbort: (() => void) | null = null;
    65→  if (userSignal) {
    66→    if (userSignal.aborted) {
    67→      clearTimeout(timer);
    68→      controller.abort(userSignal.reason);
    69→    } else {
    70→      onUserAbort = () => {
    71→        clearTimeout(timer);
    72→        controller.abort(userSignal.reason);
    73→      };
    74→      userSignal.addEventListener("abort", onUserAbort);
    75→    }
    76→  }
    77→
    78→  try {
    79→    const res = await fetch(url, {
    80→      ...init,
    81→      signal: controller.signal,
    82→    });
    83→    return res;
    84→  } catch (err: any) {
    85→    if (didTimeout) {
    86→      showToast(translate("common.error_timeout_title"), "error", {
    87→        id: "network-timeout-toast",
    88→        description: translate("common.error_network"),
    89→      });
    90→      throw new ApiError(
    91→        0,
    92→        "Request timed out. Please check your network connection.",
    93→        "TIMEOUT_ERROR",
    94→      );
    95→    }
    96→    if (userSignal?.aborted) {
    97→      throw err;
    98→    }
    99→    showToast(translate("common.error_network_title"), "error", {
   100→      id: "network-error-toast",
   101→      description: translate("common.error_network"),
   102→    });
   103→    throw new ApiError(
   104→      0,
   105→      "Network error. Unable to reach server.",
   106→      "NETWORK_ERROR",
   107→    );
   108→  } finally {
   109→    clearTimeout(timer);
   110→    if (userSignal && onUserAbort) {
   111→      userSignal.removeEventListener("abort", onUserAbort);
   112→    }
   113→  }
   114→}

   116→function getPlatform(): string {
   117→  if (typeof navigator === "undefined") return "desktop";
   118→  const ua = navigator.userAgent.toLowerCase();
   119→  if (ua.includes("win")) return "windows";
   120→  if (ua.includes("mac")) return "macos";
   121→  if (ua.includes("linux")) return "linux";
   122→  return "desktop";
   123→}

   134→function extractSubFromToken(token: string): string | undefined {
   135→  try {
   136→    const parts = token.split(".");
   137→    if (parts.length < 2) return undefined;
   138→    const base64 = parts[1].replace(/-/g, "+").replace(/_/g, "/");
   139→    const json = decodeURIComponent(
   140→      atob(base64)
   141→        .split("")
   142→        .map((c) => "%" + ("00" + c.charCodeAt(0).toString(16)).slice(-2))
   143→        .join("")
   144→    );
   145→    const payload = JSON.parse(json);
   146→    return typeof payload.sub === "string" && payload.sub.trim().length > 0 ? payload.sub.trim() : undefined;
   147→  } catch {
   148→    return undefined;
   149→  }
   150→}

   158→// upgrades youtube/google avatar artwork to maximum native resolution (=s0)
   159→export function toMaxQualityAvatarUrl(url: string | undefined): string {
   160→  if (!url) return "";
   161→  if (!/(?:[a-zA-Z0-9_-]+\.googleusercontent\.com|[a-zA-Z0-9_-]+\.ggpht\.com)/i.test(url)) {
   162→    return url;
   163→  }
   164→  if (/=w\d+-h\d+/.test(url)) {
   165→    return url.replace(/=w\d+-h\d+(?:-[a-zA-Z0-9_-]+)*/, "=s0");
   166→  }
   167→  if (/=s\d+/.test(url)) {
   168→    return url.replace(/=s\d+(?:-[a-zA-Z0-9_-]+)*/, "=s0");
   169→  }
   170→  return `${url}=s0`;
   171→}

   175→async function executeRequest<T>(
   176→  path: string,
   177→  init?: RequestInit,
   178→  allowRefresh = true,
   179→  timeoutMs = DEFAULT_TIMEOUT_MS,
   180→): Promise<T> {
   181→  const startTime = performance.now();
   182→  const method = (init?.method ?? "GET").toUpperCase();
   183→  const rawBody = typeof init?.body === "string" ? init.body : null;
   184→  const signedHeaders = await signApiRequest(method, path, rawBody);
   185→  const currentSession = getAuthSession();
   186→  let accessToken = allowRefresh ? await getValidAccessToken() : null;
   187→
   188→  if (allowRefresh && currentSession?.accessToken && !accessToken && !path.startsWith("/v1/auth/")) {
   189→    const expiryMs = getSessionExpiryMs(currentSession);
   190→    if (expiryMs && expiryMs > Date.now()) {
   191→      accessToken = currentSession.accessToken;
   192→    } else {
   193→      throw new ApiError(0, "Network error during token refresh.", "NETWORK_ERROR");
   194→    }
   195→  }
   196→
   197→  const hasBody = init?.body != null;
   198→  const extraHeaders = (init?.headers as Record<string, string>) ?? {};
   199→  const { Authorization: _ignoredAuth, ...restHeaders } = extraHeaders;
   200→
   201→  const requestId =
   202→    extraHeaders["x-request-id"] ||
   203→    extraHeaders["X-Request-Id"] ||
   204→    (typeof crypto !== "undefined" && crypto.randomUUID
   205→      ? crypto.randomUUID()
   206→      : `req_${Date.now()}_${Math.random().toString(36).slice(2, 9)}`);
   207→  const correlationHeaders: Record<string, string> = {
   208→    "x-request-id": requestId,
   209→    "x-client-version": CLIENT_VERSION,
   210→    "x-platform": getPlatform(),
   211→  };
   212→
   213→  const userId = currentSession?.user?.id || (accessToken ? extractSubFromToken(accessToken) : undefined);
   214→  if (userId) {
   215→    correlationHeaders["x-user-id"] = userId;
   216→  }
   217→  if (currentSession?.user?.username) {
   218→    correlationHeaders["x-user-name"] = currentSession.user.username;
   219→  }
   220→
   221→  const activeSessionId = getActiveSessionId();
   222→  if (activeSessionId) {
   223→    correlationHeaders["x-session-id"] = activeSessionId;
   224→  }
   225→
   226→  let res: Response;
   227→  try {
   228→    res = await fetchWithTimeout(
   229→      getApiBaseUrl() + path,
   230→      {
   231→        ...init,
   232→        headers: {
   233→          ...(hasBody ? { "Content-Type": "application/json" } : {}),
   234→          ...(accessToken ? { Authorization: `Bearer ${accessToken}` } : {}),
   235→          ...correlationHeaders,
   236→          ...signedHeaders,
   237→          ...restHeaders,
   238→        },
   239→      },
   240→      timeoutMs,
   241→    );
   242→  } catch (err: any) {
   243→    if (init?.signal?.aborted || err?.name === "AbortError") {
   244→      throw err;
   245→    }
   246→    const durationMs = Math.round(performance.now() - startTime);
   247→    telemetry.trackNetwork(method, path, 0, durationMs, requestId);
   248→    if (isConnectivityFailure(err) && !isPlaybackPath(path)) {
   249→      recordConnectivityFailure({
   250→        method,
   251→        path: stripQuery(path),
   252→        status: statusOf(err),
   253→        latencyMs: durationMs,
   254→      });
   255→    }
   256→    throw err;
   257→  }
   258→
   259→  // sync server time offset from response Date header
   260→  const dateHeader = res.headers?.get?.("date");
   261→  if (dateHeader) {
   262→    syncServerTime(dateHeader);
   263→  }
   264→
   265→  if (res.status === 401 && allowRefresh && !path.startsWith("/v1/auth/")) {
   266→    const currentSession = getAuthSession();
   267→    // If the token in session store was already rotated by another parallel request,
   268→    // retry immediately with the rotated token without firing duplicate refresh requests
   269→    let nextToken =
   270→      currentSession?.accessToken && currentSession.accessToken !== accessToken
   271→        ? currentSession.accessToken
   272→        : null;
   273→
   274→    if (!nextToken && currentSession?.refreshToken) {
   275→      const refreshed = await refreshAuthSession();
   276→      nextToken = refreshed?.accessToken ?? null;
   277→      if (!nextToken && getAuthSession()?.refreshToken) {
   278→        throw new ApiError(
   279→          0,
   280→          "Failed to refresh auth session due to network issue",
   281→          "NETWORK_ERROR",
   282→        );
   283→      }
   284→    }
   285→
   286→    if (!nextToken) {
   287→      const durationMs = Math.round(performance.now() - startTime);
   288→      telemetry.trackNetwork(method, path, res.status, durationMs, requestId);
   289→      const body = (await res.json().catch(() => ({}))) as Record<
   290→        string,
   291→        unknown
   292→      >;
   293→      const code = typeof body?.code === "string" ? body.code : undefined;
   294→      const message =
   295→        typeof body?.message === "string"
   296→          ? body.message
   297→          : typeof body?.error === "string"
   298→          ? body.error
   299→          : "Unauthorized";
   300→      throw new ApiError(res.status, message, code, requestId);
   301→    }
   302→
   303→    // Retry request with newly signed headers (synced server clock + fresh token if available)
   304→    const refreshedSignedHeaders = await signApiRequest(
   305→      method,
   306→      path,
   307→      rawBody,
   308→    );
   309→    try {
   310→      res = await fetchWithTimeout(
   311→        getApiBaseUrl() + path,
   312→        {
   313→          ...init,
   314→          headers: {
   315→            ...(hasBody ? { "Content-Type": "application/json" } : {}),
   316→            ...correlationHeaders,
   317→            ...restHeaders,
   318→            ...refreshedSignedHeaders,
   319→            ...(nextToken ? { Authorization: `Bearer ${nextToken}` } : {}),
   320→          },
   321→        },
   322→        timeoutMs,
   323→      );
   324→      const retryDateHeader = res.headers?.get?.("date");
   325→      if (retryDateHeader) {
   326→        syncServerTime(retryDateHeader);
   327→      }
   328→    } catch (err) {
   329→      const durationMs = Math.round(performance.now() - startTime);
   330→      telemetry.trackNetwork(method, path, 0, durationMs, requestId);
   331→      if (isConnectivityFailure(err) && !isPlaybackPath(path)) {
   332→        recordConnectivityFailure({
   333→          method,
   334→          path: stripQuery(path),
   335→          status: statusOf(err),
   336→          latencyMs: durationMs,
   337→        });
   338→      }
   339→      throw err;
   340→    }
   341→  }
   342→
   343→  const durationMs = Math.round(performance.now() - startTime);
   344→  telemetry.trackNetwork(method, path, res.status, durationMs, requestId);
   345→
   346→  if (!res.ok) {
   347→    const body = (await res.json().catch(() => ({}))) as Record<
   348→      string,
   349→      unknown
   350→    >;
   351→    const resRequestId =
   352→      res.headers?.get?.("x-request-id") ||
   353→      (typeof body.reqId === "string"
   354→        ? body.reqId
   355→        : typeof body.requestId === "string"
   356→        ? body.requestId
   357→        : undefined);
   358→
   359→    if (res.status >= 500) {
   360→      showToast(translate("common.error_server_title"), "error", {
   361→        id: `server-error-${res.status}`,
   362→        description: translate("common.error_server"),
   363→        requestId: resRequestId,
   364→      });
   365→    }
   366→
   367→    const apiErr = new ApiError(
   368→      res.status,
   369→      typeof body.message === "string" ? body.message : res.statusText,
   370→      typeof body.code === "string" ? body.code : undefined,
   371→      resRequestId,
   372→    );
   373→    if (isConnectivityFailure(apiErr) && !isPlaybackPath(path)) {
   374→      recordConnectivityFailure({
   375→        method,
   376→        path: stripQuery(path),
   377→        status: res.status,
   378→        ...(apiErr.code ? { code: apiErr.code } : {}),
   379→        latencyMs: durationMs,
   380→      });
   381→    }
   382→    throw apiErr;
   383→  }
   384→
   385→  if (res.status === 204) return undefined as unknown as T;
   386→  return res.json() as Promise<T>;
   387→}

   389→async function request<T>(
   390→  path: string,
   391→  init?: RequestInit,
   392→  allowRefresh = true,
   393→  timeoutMs = DEFAULT_TIMEOUT_MS,
   394→): Promise<T> {
   395→  const method = (init?.method ?? "GET").toUpperCase();
   396→  if (method === "GET" && !init?.signal) {
   397→    const key = `${path}::${allowRefresh}`;
   398→    const inFlight = inFlightGetRequests.get(key);
   399→    if (inFlight) {
   400→      return inFlight as Promise<T>;
   401→    }
   402→    const promise = executeRequest<T>(path, init, allowRefresh, timeoutMs).finally(() => {
   403→      inFlightGetRequests.delete(key);
   404→    });
   405→    inFlightGetRequests.set(key, promise);
   406→    return promise;
   407→  }
   408→  return executeRequest<T>(path, init, allowRefresh, timeoutMs);
   409→}

   411→export class ApiError extends Error {
   412→  constructor(
   413→    public readonly status: number,
   414→    message: string,
   415→    public readonly code?: string,
   416→    public readonly requestId?: string,
   417→  ) {
   418→    super(message);
   419→    this.name = "ApiError";
   420→  }

   422→  get traceInfo(): string | undefined {
   423→    return this.requestId ? `[reqId: ${this.requestId}]` : undefined;

   427→// maps any thrown error to a localized human-friendly message.
   428→// never surfaces raw api bodies, statuses, or technical jargon.
   429→export function resolveApiErrorMessage(
   430→  err: unknown,
   431→  t: (key: string) => string,
   432→  fallbackKey: string,
   433→): string {
   434→  if (err instanceof ApiError) {
   435→    if (err.status === 0) return t("common.error_network");
   436→    if (err.status === 401 || err.status === 403)
   437→      return t("common.error_session_expired");
   438→    if (err.status === 404) return t("common.error_not_found");
   439→    if (err.status === 429) return t("common.error_too_many");
   440→    if (err.status >= 500) return t("common.error_server");
   441→    return t(fallbackKey);
   442→  }
   443→  return t(fallbackKey);
   444→}

   890→  async *streamLyrics(
   891→    id: string,
   892→    signal?: AbortSignal,
   893→    hint?: { title: string; artist: string; durationMs?: number },
   894→  ): AsyncGenerator<LyricsStreamEvent> {
   895→    const hintParams =
   896→      hint?.title.trim() && hint.artist.trim()
   897→        ? new URLSearchParams({
   898→            title: hint.title.trim(),
   899→            artist: hint.artist.trim(),
   900→            ...(hint.durationMs && hint.durationMs > 0 ? { durationMs: String(Math.round(hint.durationMs)) } : {}),
   901→          }).toString()
   902→        : "";
   903→    const path = `/v1/tracks/${encodeURIComponent(id)}/lyrics${hintParams ? `?${hintParams}` : ""}`;
   904→    const signedHeaders = await signApiRequest("GET", path);
   905→    const accessToken = await getValidAccessToken();
   906→    const requestId =
   907→      typeof crypto !== "undefined" && crypto.randomUUID
   908→        ? crypto.randomUUID()
   909→        : `req_${Date.now()}_${Math.random().toString(36).slice(2, 9)}`;
   910→
   911→    const correlationHeaders: Record<string, string> = {
   912→      "x-request-id": requestId,
   913→      "x-client-version": CLIENT_VERSION,
   914→      "x-platform": getPlatform(),
   915→      "x-session-id": id,
   916→    };
   917→
   918→    let res: Response;
   919→    try {
   920→      res = await fetch(
   921→        `${getApiBaseUrl()}${path}`,
   922→        {
   923→          signal,
   924→          headers: {
   925→            ...(accessToken ? { Authorization: `Bearer ${accessToken}` } : {}),
   926→            ...correlationHeaders,
   927→            ...signedHeaders,
   928→          },
   929→        },
   930→      );
   931→    } catch (fetchErr: any) {
   932→      if (signal?.aborted || fetchErr?.name === "AbortError") {
   933→        throw fetchErr;
   934→      }
   935→      throw fetchErr;
   936→    }
   937→
   938→    const lyricsDateHeader = res.headers?.get?.("date");
   939→    if (lyricsDateHeader) {
   940→      syncServerTime(lyricsDateHeader);
   941→    }
   942→
   943→    if (!res.ok || !res.body) {
   944→      const respReqId = res.headers?.get?.("x-request-id") || requestId;
   945→      throw new ApiError(res.status, res.statusText, undefined, respReqId);
   946→    }
   947→
   948→    const reader = res.body.getReader();
   949→    const decoder = new TextDecoder();
   950→    let buffer = "";
   951→
   952→    const flushEvent = (): LyricsStreamEvent | null => {
   953→      const sep = buffer.indexOf("\n\n");
   954→      if (sep === -1) return null;
   955→      const rawEvent = buffer.slice(0, sep);
   956→      buffer = buffer.slice(sep + 2);
   957→
   958→      const dataLines: string[] = [];
   959→      for (const line of rawEvent.split("\n")) {
   960→        if (line.startsWith("data:")) {
   961→          dataLines.push(line.slice(5).trim());
   962→        }
   963→      }
   964→
   965→      if (dataLines.length === 0) return null;
   966→      try {
   967→        return JSON.parse(dataLines.join("\n")) as LyricsStreamEvent;
   968→      } catch {
   969→        return null;
   970→      }
   971→    };
   972→
   973→    try {
   974→      while (true) {
   975→        const { done, value } = await reader.read();
   976→        if (done) break;
   977→        buffer += decoder.decode(value, { stream: true });
   978→
   979→        let event = flushEvent();
   980→        while (event) {
   981→          yield event;
   982→          event = flushEvent();
   983→        }
   984→      }
   985→
   986→      buffer += decoder.decode();
   987→      let event = flushEvent();
   988→      while (event) {
   989→        yield event;
   990→        event = flushEvent();
   991→      }
   992→    } finally {
   993→      reader.releaseLock();
   994→    }
   995→  },
