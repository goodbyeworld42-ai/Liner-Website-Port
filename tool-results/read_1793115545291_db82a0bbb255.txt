    29→// error recovery tuning (see the audio "error" handler)
    30→const RECOVERY_HOLD_MS = 3000;
    31→const WAITING_GRACE_MS = 300; // a buffering hiccup shorter than this never shows // must play this far past a recovery before it may retry again

    33→// A stuck load must never sit in "loading" forever with no feedback (the
    34→// "spinner spins but no error ever appears" failure mode). Both stream
    35→// resolution and the audio.play() call get a deadline; when one fires we
    36→// abort the load and surface a real error (toast + auto-skip + diagnostics)
    37→// instead of hanging silently.

    41→export class PlaybackTimeoutError extends Error {
    42→  constructor(phase: "resolve" | "start", trackTitle: string, ms: number) {
    43→    super(
    44→      phase === "resolve"
    45→        ? `Timed out resolving audio for "${trackTitle}" after ${Math.round(ms / 1000)}s.`
    46→        : `Audio never started for "${trackTitle}" within ${Math.round(ms / 1000)}s.`,
    47→    );
    48→    this.name = "PlaybackTimeoutError";
    49→  }

   145→  private setupDeviceChangeListener() {
   146→    if (typeof navigator === "undefined" || !navigator.mediaDevices) return;
   147→
   148→    let initial = true;
   149→    const onDeviceChange = () => {
   150→      if (initial) {
   151→        initial = false;

   153→      }
   154→      if (!usePlayerStore.getState().pauseOnDeviceChange) return;
   155→      if (
   156→        !this.audio.paused &&
   157→        usePlayerStore.getState().status === "playing"

   164→      navigator.mediaDevices.addEventListener("devicechange", onDeviceChange);
   165→      void navigator.mediaDevices.enumerateDevices().then(() => {
   166→        initial = false;
   167→      });

   173→  private setupMediaSession() {
   174→    if (typeof navigator === "undefined" || !("mediaSession" in navigator))
   175→      return;
   176→
   177→    try {
   178→      navigator.mediaSession.setActionHandler("play", () => {
   179→        usePlayerStore.getState().dispatch({ type: "RESUME_REQUESTED" });
   180→      });
   181→      navigator.mediaSession.setActionHandler("pause", () => {
   182→        usePlayerStore.getState().dispatch({ type: "PAUSE_REQUESTED" });
   183→      });
   184→      navigator.mediaSession.setActionHandler("previoustrack", () => {
   185→        usePlayerStore.getState().dispatch({ type: "SKIP_PREVIOUS" });
   186→      });
   187→      navigator.mediaSession.setActionHandler("nexttrack", () => {
   188→        usePlayerStore.getState().dispatch({ type: "SKIP_NEXT" });
   189→      });
   190→      navigator.mediaSession.setActionHandler("stop", () => {
   191→        usePlayerStore.getState().dispatch({ type: "PAUSE_REQUESTED" });
   192→      });
   193→      const seekBy = (deltaSeconds: number) => {
   194→        const st = usePlayerStore.getState();
   195→        const max =
   196→          st.durationMs > 0 ? st.durationMs : Number.POSITIVE_INFINITY;
   197→        const next = Math.min(
   198→          max,
   199→          Math.max(0, st.positionMs + deltaSeconds * 1000),
   200→        );
   201→        st.dispatch({ type: "SEEK_COMMITTED", positionMs: Math.round(next) });
   202→      };
   203→      navigator.mediaSession.setActionHandler("seekbackward", (details) => {
   204→        seekBy(-(details.seekOffset ?? 10));
   205→      });
   206→      navigator.mediaSession.setActionHandler("seekforward", (details) => {
   207→        seekBy(details.seekOffset ?? 10);
   208→      });
   209→      navigator.mediaSession.setActionHandler("seekto", (details) => {
   210→        if (details.seekTime !== undefined && !Number.isNaN(details.seekTime)) {
   211→          usePlayerStore.getState().dispatch({
   212→            type: "SEEK_COMMITTED",
   213→            positionMs: Math.round(details.seekTime * 1000),
   214→          });
   215→        }
   216→      });
   217→    } catch {
   218→      // ignore unsupported media session actions
   219→    }
   220→  }

   222→  // windows' media controls (and apps built on them, like FluentFlyout) read this.
   223→  // "loading" between tracks still reports playing so the session never looks finished.
   224→  private syncMediaSessionState(status: string) {
   225→    if (typeof navigator === "undefined" || !("mediaSession" in navigator))
   226→      return;
   227→    try {
   228→      navigator.mediaSession.playbackState =
   229→        status === "playing" || status === "loading"
   230→          ? "playing"
   231→          : status === "paused"
   232→            ? "paused"
   233→            : "none";
   234→      if (status === "idle") navigator.mediaSession.metadata = null;
   235→    } catch {
   236→      // ignore unsupported environments
   237→    }
   238→  }
   239→
   240→  // progress / seek bar in the system controls
   241→  private syncMediaSessionPosition() {
   242→    if (typeof navigator === "undefined" || !("mediaSession" in navigator))
   243→      return;
   244→    const ms = navigator.mediaSession as MediaSession & {
   245→      setPositionState?: (state?: MediaPositionState) => void;
   246→    };
   247→    if (typeof ms.setPositionState !== "function") return;
   248→    if (this.attachedTrackId !== this.currentTrackId) return;
   249→    const duration = this.audio.duration;
   250→    if (!Number.isFinite(duration) || duration <= 0) return;
   251→    try {
   252→      ms.setPositionState({
   253→        duration,
   254→        playbackRate: this.audio.playbackRate > 0 ? this.audio.playbackRate : 1,
   255→        position: Math.min(duration, Math.max(0, this.audio.currentTime || 0)),
   256→      });
   257→    } catch {
   258→      // ignore invalid transient states
   259→    }
   260→  }

   288→  private setupAudioListeners(audioEl: HTMLAudioElement) {
   289→    audioEl.addEventListener("timeupdate", () => {
   290→      if (this.isResetting || audioEl !== this.audio) return;
   291→      if (!this.currentTrackId) return;
   292→      if (usePlayerStore.getState().status === "loading") return;
   293→      if (Number.isNaN(audioEl.currentTime)) return;
   294→
   295→      const posMs = Math.round(audioEl.currentTime * 1000);
   296→      usePlayerStore.getState().setPosition(posMs);
   297→      if (
   298→        this.recoveryHoldUntilMs !== null &&
   299→        posMs >= this.recoveryHoldUntilMs
   300→      ) {
   301→        this.recoveryHoldUntilMs = null;
   302→        this.recoveryAttempts = 0;
   303→      }
   304→    });
   305→
   306→    audioEl.addEventListener("durationchange", () => {
   307→      if (this.isResetting || audioEl !== this.audio) return;
   308→      if (!this.currentTrackId) return;
   309→      if (Number.isNaN(audioEl.duration) || audioEl.duration <= 0) return;
   310→
   311→      usePlayerStore
   312→        .getState()
   313→        .setDuration(Math.round(audioEl.duration * 1000));
   314→      this.syncMediaSessionPosition();
   315→    });
   316→
   317→    for (const type of [
   318→      "seeked",
   319→      "ratechange",
   320→      "play",
   321→      "pause",
   322→      "loadedmetadata",
   323→    ]) {
   324→      audioEl.addEventListener(type, () => {
   325→        if (this.isResetting || audioEl !== this.audio) return;
   326→        this.syncMediaSessionPosition();
   327→      });
   328→    }
   329→
   330→    audioEl.addEventListener("waiting", () => {
   331→      if (this.isResetting || audioEl !== this.audio) return;
   332→      log("yellow", "audio", "buffering (waiting)");
   333→      this.stallStartTime = performance.now();
   334→      if (this.waitingTimer) clearTimeout(this.waitingTimer);
   335→      this.waitingTimer = setTimeout(() => {
   336→        this.waitingTimer = null;
   337→        // still stuck: a real stall, show it
   338→        if (audioEl !== this.audio || audioEl.paused || audioEl.readyState >= 3)
   339→          return;
   340→        if (usePlayerStore.getState().status === "playing") {
   341→          usePlayerStore.getState().setStatus("loading");
   342→        }
   343→      }, WAITING_GRACE_MS);
   344→    });
   345→
   346→    // resumed (or never really stalled): drop a pending "loading"
   347→    for (const type of ["playing", "seeked", "canplay", "pause", "emptied"]) {
   348→      audioEl.addEventListener(type, () => {
   349→        if (audioEl !== this.audio || !this.waitingTimer) return;
   350→        if (type === "seeked" && audioEl.readyState < 3) return; // seeked but still no data
   351→        clearTimeout(this.waitingTimer);
   352→        this.waitingTimer = null;
   353→      });
   354→    }
   355→
   356→    audioEl.addEventListener("ended", () => {
   357→      if (this.isResetting || audioEl !== this.audio) return;
   358→      if (!this.currentTrackId) return;
   359→      if (this.attachedTrackId !== this.currentTrackId) return;
   360→
   361→      // Ensure the track truly finished playing (not an empty media load or teardown artefact)
   362→      const duration = audioEl.duration;
   363→      const currentTime = audioEl.currentTime;
   364→      if (duration > 0 && currentTime >= Math.max(0, duration - 1.5)) {
   365→        log("cyan", "audio", "ended — skipping to next");
   366→        this.onEnded?.();
   367→        const state = usePlayerStore.getState();
   368→        state.dispatch({ type: "SKIP_NEXT", isAutoEnd: true });
   369→      }
   370→    });
   371→
   372→    audioEl.addEventListener("playing", () => {
   373→      if (this.isResetting || audioEl !== this.audio) return;
   374→      log("green", "audio", "playing");
   375→      if (this.recoveryHoldUntilMs === null) this.recoveryAttempts = 0;
   376→      this.applyPlaybackRateAndPitch();
   377→
   378→      if (this.stallStartTime !== null) {
   379→        const stallDurationMs = Math.round(
   380→          performance.now() - this.stallStartTime,
   381→        );
   382→        if (stallDurationMs > 400) {
   383→          telemetry.trackPlaybackHealth("STALL_BUFFERING", {
   384→            trackId: this.currentTrackId || undefined,
   385→            stallDurationMs,
   386→          });
   387→        }
   388→        this.stallStartTime = null;
   389→      }
   390→
   391→      if (!this.currentTrackId || audioEl.paused) return;
   392→      const currentStatus = usePlayerStore.getState().status;
   393→      if (currentStatus !== "paused") {
   394→        usePlayerStore.getState().setStatus("playing");
   395→      }
   396→    });
   397→
   398→    audioEl.addEventListener("pause", () => {
   399→      if (this.isResetting || audioEl !== this.audio) return;
   400→      log("cyan", "audio", "paused");
   401→      const st = usePlayerStore.getState().status;
   402→      if (st === "playing") {
   403→        usePlayerStore.getState().setStatus("paused");
   404→      }
   405→    });
   406→
   407→    audioEl.addEventListener("error", () => {
   408→      if (this.isResetting || audioEl !== this.audio) return;
   409→      if (
   410→        !this.currentTrackId ||
   411→        !audioEl.src ||
   412→        audioEl.src === window.location.href
   413→      )
   414→        return;
   415→      // the previous track's source is still attached while the next one loads
   416→      if (this.attachedTrackId !== this.currentTrackId) return;
   417→
   418→      const message = audioEl.error?.message || "Audio playback error";
   419→      const code = audioEl.error?.code;
   420→      console.error(
   421→        "[AudioElementError]",
   422→        this.currentTrackId,
   423→        code,
   424→        message,
   425→        audioEl.src,
   426→      );
   427→      log("red", "audio", `error (code ${code}): ${message}`);
   428→
   429→      const isDecode =
   430→        code ===
   431→          (typeof MediaError !== "undefined"
   432→            ? MediaError.MEDIA_ERR_DECODE
   433→            : 3) ||
   434→        code ===
   435→          (typeof MediaError !== "undefined"
   436→            ? MediaError.MEDIA_ERR_SRC_NOT_SUPPORTED
   437→            : 4);
   438→
   439→      if (isDecode) {
   440→        telemetry.trackPlaybackHealth("MEDIA_DECODE_ERROR", {
   441→          trackId: this.currentTrackId || undefined,
   442→          errorMessage: message,
   443→          details: { mediaErrorCode: code },
   444→        });
   445→      }
   446→
   447→      if (this.attachedFromLiner) {
   448→        this.attachedFromLiner = false;
   449→        markLinerStreamFailed(`media error ${code}`);
   450→      }
   451→
   452→      if (this.activeBlobUrl) {
   453→        URL.revokeObjectURL(this.activeBlobUrl);
   454→        this.activeBlobUrl = null;
   455→        void linerDb.deleteAudio(this.currentTrackId);
   456→        audioCache.deleteAudio(this.currentTrackId);
   457→      }
   458→
   459→      const store = usePlayerStore.getState();
   460→      const currentTrack = store.currentTrack;
   461→      const posMs =
   462→        Number.isFinite(audioEl.currentTime) && audioEl.currentTime > 0
   463→          ? Math.round(audioEl.currentTime * 1000)
   464→          : store.positionMs;
   465→
   466→      if (
   467→        currentTrack &&
   468→        currentTrack.id === this.currentTrackId &&
   469→        this.recoveryAttempts < 1
   470→      ) {
   471→        this.recoveryAttempts++;
   472→        this.recoveryHoldUntilMs = posMs + RECOVERY_HOLD_MS;
   473→        // the first resolution already produced a broken stream — the retry
   474→        // must change paths (skip the Liner r2 probe) or it replays the same
   475→        // failure; resolvePlaybackStream's SoundCloud rescue still applies
   476→        this.forceFallbackResolve = true;
   477→        log(
   478→          "yellow",
   479→          "playback",
   480→          `audio error encountered — recovering via cache/stream at ${posMs}ms`,
   481→        );
   482→        void this.loadAndPlay(currentTrack, posMs, true);
   483→        return;
   484→      }
   485→
   486→      usePlayerStore.getState().setStatus("error", message);
   487→      this.onError?.({ trackId: this.currentTrackId, message });
   488→    });
   489→  }

   495→  public hasLoadedSourceFor(trackId: string): boolean {
   496→    const isExpired =
   497→      this.currentSessionExpiresAt !== null &&
   498→      Date.now() >= this.currentSessionExpiresAt - 15_000;
   499→    return (
   500→      this.currentTrackId === trackId &&
   501→      this.attachedTrackId === trackId &&
   502→      Boolean(this.audio.currentSrc || this.audio.src) &&
   503→      this.audio.src !== window.location.href &&
   504→      !this.audio.error &&
   505→      !isExpired
   506→    );
   507→  }

   509→  public applyPlaybackRateAndPitch(
   510→    rate?: number,
   511→    preservesPitch?: boolean,
   512→  ): void {
   513→    if (!this.audio) return;
   514→    const store = usePlayerStore.getState();
   515→    const targetRate = rate ?? store.playbackRate ?? 1;
   516→    const targetPreserves =
   517→      preservesPitch ??
   518→      (store.isPitchLinked !== undefined ? !store.isPitchLinked : true);
   519→
   520→    try {
   521→      this.audio.playbackRate = targetRate;
   522→      this.audio.defaultPlaybackRate = targetRate;
   523→      this.audio.preservesPitch = targetPreserves;
   524→      (this.audio as any).mozPreservesPitch = targetPreserves;
   525→      (this.audio as any).webkitPreservesPitch = targetPreserves;
   526→    } catch {
   527→      // Audio element not initialized or properties unsupported
   528→    }
   529→
   530→    try {
   531→      const isPitchLinked = store.isPitchLinked ?? true;
   532→      const pitchSemitones = store.pitchSemitones ?? 0;
   533→      const isReverbEnabled = store.isReverbEnabled ?? false;
   534→      const reverbLevel = store.reverbLevel ?? 0.35;
   535→      const needsEffects =
   536→        isReverbEnabled ||
   537→        (!isPitchLinked && Math.abs(pitchSemitones) >= 0.05) ||
   538→        isEqActive();
   539→      if (!needsEffects && !pitchShifter.isAttached()) return;
   540→      pitchShifter.attach(this.audio);
   541→      pitchShifter.update({
   542→        pitchSemitones,
   543→        isPitchLinked,
   544→        isReverbEnabled,
   545→        reverbLevel,
   546→      });
   547→      // re-assert EQ after (re)attach so a fresh graph serves the saved curve
   548→      ensureEqualizerApplied();
   549→    } catch (err) {
   550→      console.error("[PlayerRuntime] applyPlaybackRateAndPitch error:", err);
   551→    }
   552→  }

   561→  private fadeVolume(
   562→    targetVolume: number,
   563→    durationMs: number = 180,
   564→    onComplete?: () => void,
   565→  ): void {
   566→    this.cancelFade();
   567→    const startVolume = this.audio.volume;
   568→    const diff = targetVolume - startVolume;
   569→    if (Math.abs(diff) < 0.001 || durationMs <= 0) {
   570→      this.audio.volume = Math.max(0, Math.min(1, targetVolume));
   571→      onComplete?.();
   572→      return;
   573→    }
   574→
   575→    const startTime = performance.now();
   576→
   577→    const tick = () => {
   578→      const now = performance.now();
   579→      const elapsed = now - startTime;
   580→      const progress = Math.min(elapsed / durationMs, 1);
   581→      const eased = 0.5 - 0.5 * Math.cos(progress * Math.PI);
   582→      const current = startVolume + diff * eased;
   583→      this.audio.volume = Math.max(0, Math.min(1, current));
   584→
   585→      if (progress < 1) {
   586→        // audio fade ticks independently of display vsync in background
   587→        this.fadeTimer = setTimeout(tick, 16);
   588→      } else {
   589→        this.fadeTimer = null;
   590→        this.audio.volume = Math.max(0, Math.min(1, targetVolume));
   591→        onComplete?.();
   592→      }
   593→    };
   594→
   595→    this.fadeTimer = setTimeout(tick, 16);
   596→  }
   597→
   598→  private releaseStream() {
   599→    this.releaseActiveStream?.();
   600→    this.releaseActiveStream = null;
   601→  }
   602→
   603→  private watchFullSource(trackId: string, stream: PlaybackStream) {
   604→    const { completed, needsFullSource } = stream;
   605→    if (!completed || !needsFullSource) return;
   606→    void completed
   607→      .then((blob) => {
   608→        if (
   609→          this.currentTrackId !== trackId ||
   610→          this.audio.src !== rewriteForWeb(stream.url)
   611→        )
   612→          return;
   613→        if (needsFullSource()) this.swapToFullSource(blob);
   614→      })
   615→      .catch(() => undefined);
   616→  }
   617→
   618→  private swapToFullSource(blob: Blob) {
   619→    const position = this.audio.currentTime;
   620→    const wasPlaying = !this.audio.paused;
   621→    const url = URL.createObjectURL(blob);
   622→    this.isResetting = true;
   623→    this.releaseStream();
   624→    if (this.activeBlobUrl) URL.revokeObjectURL(this.activeBlobUrl);
   625→    this.activeBlobUrl = url;
   626→    this.audio.src = url;
   627→    this.applyPlaybackRateAndPitch();
   628→    const settle = () => {
   629→      this.isResetting = false;
   630→    };
   631→    this.audio.addEventListener(
   632→      "loadedmetadata",
   633→      () => {
   634→        this.audio.currentTime = position;
   635→        if (wasPlaying) void this.audio.play().then(settle, settle);
   636→        else settle();
   637→      },
   638→      { once: true },
   639→    );
   640→    log("cyan", "playback", "switched to the fully downloaded source");
   641→  }

   643→  private safeResetAudioElement() {
   644→    this.isResetting = true;
   645→    this.currentSessionExpiresAt = null;
   646→    try {
   647→      this.cancelFade();
   648→      this.audio.pause();
   649→      this.releaseStream();
   650→      if (this.activeBlobUrl) {
   651→        URL.revokeObjectURL(this.activeBlobUrl);
   652→        this.activeBlobUrl = null;
   653→      }
   654→      this.audio.removeAttribute("src");
   655→      this.audio.load();
   656→      this.attachedTrackId = null;
   657→    } catch {
   658→    } finally {
   659→      this.isResetting = false;
   660→    }
   661→  }

   663→  // between tracks: silence the old source but keep it attached until the next one
   664→  // is ready. Emptying the element here made Chromium drop the system media session
   665→  // during every skip, so Windows controls / FluentFlyout lost Liner after a track change.
   666→  private pauseForTransition() {
   667→    this.isResetting = true;
   668→    this.currentSessionExpiresAt = null;
   669→    try {
   670→      this.cancelFade();
   671→      this.audio.pause();
   672→    } catch {
   673→    } finally {
   674→      this.isResetting = false;
   675→    }
   676→  }

   678→  private attachSource(url: string, isBlob: boolean, trackId: string) {
   679→    const previousBlob = this.activeBlobUrl;
   680→    // web port: route remote media through the same-origin media proxy so the
   681→    // crossOrigin="anonymous" audio element stays readable for the visualizer
   682→    this.audio.src = rewriteForWeb(url);
   683→    this.activeBlobUrl = isBlob ? url : null;
   684→    this.attachedTrackId = trackId;
   685→    this.attachedFromLiner = false;
   686→    if (previousBlob && previousBlob !== url) URL.revokeObjectURL(previousBlob);
   687→  }

   694→    this.backgroundCacheAbortController = null;
   695→    this.currentTrackId = null;
   696→    this.currentSessionExpiresAt = null;
   697→    this.recoveryAttempts = 0;
   698→    this.recoveryHoldUntilMs = null;

   704→  public async preloadTrack(track: Track): Promise<void> {
   705→    if (this.currentTrackId === track.id) return;
   706→    if (track.localFileId) return; // local blobs are already "cached"
   707→    if (this.preloadingTrackId === track.id) return;
   708→
   709→    const cached = await linerDb.getAudio(track.id);
   710→    if (cached) return;
   711→
   712→    this.preloadAbortController?.abort();
   713→    const controller = new AbortController();
   714→    this.preloadAbortController = controller;
   715→    this.preloadingTrackId = track.id;
   716→
   717→    const generation = audioCache.currentGeneration();
   718→    const promise = (async () => {
   719→      try {
   720→        const stream = await resolvePlaybackStream(track, controller.signal);
   721→        if (controller.signal.aborted) return;
   722→
   723→        const blob = await this.downloadAudioBlob(stream, controller.signal);
   724→        if (!blob || controller.signal.aborted) return;
   725→        if (
   726→          !(await audioCache.saveAudioBlob(
   727→            track.id,
   728→            blob,
   729→            blob.type,
   730→            generation,
   731→          ))
   732→        )
   733→          return;
   734→        await linerDb.putTrack(track);
   735→        log(
   736→          "green",
   737→          "preload",
   738→          `preloaded audio: "${track.title}" (${Math.round(blob.size / 1024)} KB)`,
   739→        );
   740→      } catch {
   741→      } finally {
   742→        if (this.preloadingTrackId === track.id) {
   743→          this.preloadingTrackId = null;
   744→          this.preloadingPromise = null;
   745→          this.preloadAbortController = null;
   746→        }
   747→      }
   748→    })();
   749→
   750→    this.preloadingPromise = promise;
   751→    await promise;
   752→  }

   788→  public async loadAndPlay(
   789→    track: Track,
   790→    startPositionMs: number = 0,
   791→    bypassCache: boolean = false,
   792→  ) {
   793→    this.cancelFade();
   794→    this.pendingStartPositionMs = startPositionMs;
   795→    const epoch = ++this.loadEpoch;
   796→
   797→    if (this.currentTrackId !== track.id) {
   798→      this.recoveryAttempts = 0;
   799→      this.recoveryHoldUntilMs = null;
   800→      this.forceFallbackResolve = false;
   801→    }
   802→
   803→    if (this.preloadingTrackId === track.id && this.preloadingPromise) {
   804→      await this.preloadingPromise;
   805→    } else if (this.preloadingTrackId && this.preloadingTrackId !== track.id) {
   806→      this.preloadAbortController?.abort();
   807→      this.preloadAbortController = null;
   808→      this.preloadingTrackId = null;
   809→      this.preloadingPromise = null;
   810→    }
   811→
   812→    this.loadAbortController?.abort();
   813→    const controller = new AbortController();
   814→    this.loadAbortController = controller;
   815→
   816→    this.backgroundCacheAbortController?.abort();
   817→    const bgCacheController = new AbortController();
   818→    this.backgroundCacheAbortController = bgCacheController;
   819→
   820→    const store = usePlayerStore.getState();
   821→    if (!store.keepSpeedAcrossTracks && this.currentTrackId !== track.id) {
   822→      store.resetSpeedAndPitch();
   823→    }
   824→    this.currentTrackId = track.id;
   825→    this.loadStartTime = performance.now();
   826→    this.updateMediaSessionMetadata(track);
   827→
   828→    store.setStatus("loading", null);
   829→    store.setPosition(startPositionMs > 0 ? startPositionMs : 0);
   830→
   831→    this.pauseForTransition();
   832→
   833→    try {
   834→      const cachedAudio = bypassCache ? null : await linerDb.getAudio(track.id);
   835→      if (epoch !== this.loadEpoch || controller.signal.aborted) {
   836→        return;
   837→      }
   838→
   839→      const isValidCachedBlob =
   840→        Boolean(cachedAudio?.blob) &&
   841→        cachedAudio!.blob.size > 0 &&
   842→        (!cachedAudio!.blob.type ||
   843→          cachedAudio!.blob.type.startsWith("audio/"));
   844→
   845→      if (cachedAudio && !isValidCachedBlob) {
   846→        void linerDb.deleteAudio(track.id);
   847→        audioCache.deleteAudio(track.id);
   848→      }
   849→
   850→      if (cachedAudio && isValidCachedBlob) {
   851→        this.currentSessionExpiresAt = null;
   852→        void linerDb.putTrack(track);
   853→        const blobUrl = URL.createObjectURL(cachedAudio.blob);
   854→        this.attachSource(blobUrl, true, track.id);
   855→        this.audio.volume = 0;
   856→        this.audio.load();
   857→        this.applyPlaybackRateAndPitch();
   858→
   859→        const targetPosition =
   860→          this.pendingStartPositionMs > 0
   861→            ? this.pendingStartPositionMs
   862→            : startPositionMs;
   863→        this.seekWhenReady(epoch, targetPosition);
   864→        log(
   865→          "cyan",
   866→          "playback",
   867→          `audio cache hit for "${track.title}" (${Math.round(cachedAudio.byteSize / 1024)} KB)`,
   868→        );
   869→
   870→        if (this.audio.readyState < 3) {
   871→          await new Promise<void>((resolve) => {
   872→            const timeout = setTimeout(resolve, 3000);
   873→            const onCanPlay = () => {
   874→              clearTimeout(timeout);
   875→              this.audio.removeEventListener("canplay", onCanPlay);
   876→              this.audio.removeEventListener("error", onCanPlay);
   877→              resolve();
   878→            };
   879→            this.audio.addEventListener("canplay", onCanPlay, { once: true });
   880→            this.audio.addEventListener("error", onCanPlay, { once: true });
   881→          });
   882→        }
   883→
   884→        if (epoch !== this.loadEpoch || controller.signal.aborted) return;
   885→
   886→        const playPromise = this.audio.play();
   887→        this.activePlayPromise = playPromise;
   888→        await withDeadline(
   889→          playPromise,
   890→          PLAY_START_DEADLINE_MS,
   891→          () =>
   892→            new PlaybackTimeoutError(
   893→              "start",
   894→              track.title,
   895→              PLAY_START_DEADLINE_MS,
   896→            ),
   897→        );
   898→
   899→        const audioStartLatencyMs = Math.round(
   900→          performance.now() - this.loadStartTime,
   901→        );
   902→
   903→        if (epoch === this.loadEpoch) {
   904→          usePlayerStore.getState().setStatus("playing");
   905→          this.fadeVolume(usePlayerStore.getState().volume, 180);
   906→        }
   907→        log(
   908→          "green",
   909→          "playback",
   910→          `started (cached): ${track.title} in ${audioStartLatencyMs}ms`,
   911→        );
   912→        return;
   913→      }
   914→
   915→      const stream = await withDeadline(
   916→        resolvePlaybackStream(track, controller.signal, {
   917→          progressive: true,
   918→          skipLinerCache: this.forceFallbackResolve,
   919→        }),
   920→        RESOLVE_DEADLINE_MS,
   921→        () =>
   922→          new PlaybackTimeoutError("resolve", track.title, RESOLVE_DEADLINE_MS),
   923→      );
   924→      this.forceFallbackResolve = false;
   925→      this.ttfbMs = Math.round(performance.now() - this.loadStartTime);
   926→
   927→      if (epoch !== this.loadEpoch || controller.signal.aborted) {
   928→        stream.release?.();
   929→        log(
   930→          "yellow",
   931→          "playback",
   932→          `superseded before stream ready (${track.title})`,
   933→        );
   934→        return;
   935→      }
   936→
   937→      const streamCodec = stream.codec;
   938→      const streamBitrate = stream.bitrate;
   939→      this.currentSessionExpiresAt = stream.expiresAtMs;
   940→      log("cyan", "playback", `resolved stream for ${track.title}`);
   941→      this.releaseStream();
   942→      const url = stream.blob ? URL.createObjectURL(stream.blob) : stream.url;
   943→      this.attachSource(url, Boolean(stream.blob), track.id);
   944→      this.attachedFromLiner = Boolean(stream.fromLiner);
   945→      this.releaseActiveStream = stream.release ?? null;
   946→      this.watchFullSource(track.id, stream);
   947→      this.audio.volume = 0;
   948→      this.audio.load();
   949→      this.applyPlaybackRateAndPitch();
   950→
   951→      const targetPosition =
   952→        this.pendingStartPositionMs > 0
   953→          ? this.pendingStartPositionMs
   954→          : startPositionMs;
   955→      this.seekWhenReady(epoch, targetPosition);
   956→      log(
   957→        "green",
   958→        "playback",
   959→        `stream attached: ${streamCodec} ${streamBitrate} bps (${track.title})`,
   960→      );
   961→
   962→      if (this.audio.readyState < 3) {
   963→        await new Promise<void>((resolve) => {
   964→          const timeout = setTimeout(resolve, 3000);
   965→          const onCanPlay = () => {
   966→            clearTimeout(timeout);
   967→            this.audio.removeEventListener("canplay", onCanPlay);
   968→            this.audio.removeEventListener("error", onCanPlay);
   969→            resolve();
   970→          };
   971→          this.audio.addEventListener("canplay", onCanPlay, { once: true });
   972→          this.audio.addEventListener("error", onCanPlay, { once: true });
   973→        });
   974→      }
   975→
   976→      if (epoch !== this.loadEpoch || controller.signal.aborted) return;
   977→
   978→      const playPromise = this.audio.play();
   979→      this.activePlayPromise = playPromise;
   980→      await withDeadline(
   981→        playPromise,
   982→        PLAY_START_DEADLINE_MS,
   983→        () =>
   984→          new PlaybackTimeoutError("start", track.title, PLAY_START_DEADLINE_MS),
   985→      );
   986→
   987→      const audioStartLatencyMs = Math.round(
   988→        performance.now() - this.loadStartTime,
   989→      );
   990→
   991→      if (epoch === this.loadEpoch) {
   992→        usePlayerStore.getState().setStatus("playing");
   993→        this.fadeVolume(usePlayerStore.getState().volume, 180);
   994→        this.cacheAudioInBackground(track, stream, bgCacheController.signal);
   995→        // website-port play feed for the owner's User Finder (fire-and-forget)
   996→        reportWebsitePlay(track);
   997→      }
   998→      log(
   999→        "green",
  1000→        "playback",
  1001→        `started: ${track.title} in ${audioStartLatencyMs}ms`,
  1002→      );
  1003→
  1004→      telemetry.trackPlaybackHealth("INIT_LATENCY", {
  1005→        trackId: track.id,
  1006→        codec: streamCodec,
  1007→        bitrate: streamBitrate,
  1008→        timeToFirstByteMs: this.ttfbMs,
  1009→        audioStartLatencyMs,
  1010→      });
  1011→    } catch (error) {
  1012→      if (epoch !== this.loadEpoch || controller.signal.aborted) {
  1013→        return;
  1014→      }
  1015→
  1016→      if (error instanceof PlaybackTimeoutError) {
  1017→        // deadline fired while something was still pending — stop the
  1018→        // underlying work and record it so the Settings diagnostics panel
  1019→        // shows WHY this track failed instead of a silent hang
  1020→        controller.abort();
  1021→        recordPlaybackFailure("unresolved", error, track.title);
  1022→      }
  1023→
  1024→      if (error instanceof DOMException && error.name === "AbortError") {
  1025→        return;
  1026→      }
  1027→
  1028→      if (error instanceof DOMException && error.name === "NotAllowedError") {
  1029→        log(
  1030→          "yellow",
  1031→          "playback",
  1032→          "autoplay blocked — waiting for user gesture",
  1033→        );
  1034→        usePlayerStore.getState().setStatus("paused");
  1035→        return;
  1036→      }
  1037→
  1038→      if (
  1039→        isBotCheckError(error) &&
  1040→        !isYoutubeSignedIn() &&
  1041→        !isYoutubeSignInDeclined() &&
  1042→        prefersSignInPrompt()
  1043→      ) {
  1044→        log("yellow", "playback", "youtube sign-in required, pausing");
  1045→        usePlayerStore.getState().setStatus("paused");
  1046→        requestYoutubeSignIn();
  1047→        return;
  1048→      }
  1049→
  1050→      const message =
  1051→        error instanceof ApiError
  1052→          ? error.message
  1053→          : error instanceof Error
  1054→            ? error.message
  1055→            : "Playback failed to start.";
  1056→      console.error("[LoadAndPlayError]", track.id, error);
  1057→      log("red", "playback", `failed: ${message}`);
  1058→      usePlayerStore.getState().setStatus("error", message);
  1059→
  1060→      telemetry
  1061→        .reportError(error, {
  1062→          action: "playback_failed",
  1063→          trackId: track.id,
  1064→          title: track.title,
  1065→        })
  1066→        .catch(() => {});
  1067→
  1068→      this.onError?.({ trackId: track.id, message });
  1069→    } finally {
  1070→      if (epoch === this.loadEpoch) {
  1071→        this.activePlayPromise = null;
  1072→      }
  1073→    }
  1074→  }

  1076→  private seekWhenReady(epoch: number, positionMs: number) {
  1077→    if (positionMs <= 0) return;
  1078→    const apply = () => {
  1079→      if (epoch !== this.loadEpoch) return;
  1080→      try {
  1081→        this.audio.currentTime = positionMs / 1000;
  1082→      } catch {
  1083→        // Media not seekable yet — ignore
  1084→      }
  1085→    };
  1086→    if (this.audio.readyState >= 1) {
  1087→      apply();
  1088→    } else {
  1089→      this.audio.addEventListener("loadedmetadata", apply, { once: true });
  1090→    }
  1091→  }

  1106→  public resume() {
  1107→    const epoch = ++this.loadEpoch;
  1108→    this.loadAbortController?.abort();
  1109→    this.loadAbortController = null;
  1110→
  1111→    const targetVolume = usePlayerStore.getState().volume;
  1112→    usePlayerStore.getState().setStatus("playing");
  1113→    pitchShifter.resumeContext();
  1114→    this.applyPlaybackRateAndPitch();
  1115→    log("cyan", "audio", "resume requested (fading in)");
  1116→
  1117→    const currentTrack = usePlayerStore.getState().currentTrack;
  1118→    const isExpired =
  1119→      this.currentSessionExpiresAt !== null &&
  1120→      Date.now() >= this.currentSessionExpiresAt - 15_000;
  1121→    const hasMediaError = Boolean(this.audio.error);
  1122→    const hasValidSource =
  1123→      Boolean(this.audio.src) &&
  1124→      this.audio.src !== window.location.href &&
  1125→      !isExpired &&
  1126→      !hasMediaError;
  1127→
  1128→    if (!hasValidSource && currentTrack) {
  1129→      void this.loadAndPlay(currentTrack, usePlayerStore.getState().positionMs);
  1130→      return;
  1131→    }
  1132→
  1133→    if (this.audio.paused) {
  1134→      this.audio.volume = 0;
  1135→      const playPromise = this.audio.play();
  1136→      this.activePlayPromise = playPromise;
  1137→      playPromise
  1138→        .then(() => {
  1139→          if (epoch !== this.loadEpoch) return;
  1140→          this.fadeVolume(targetVolume, 180);
  1141→        })
  1142→        .catch((error) => {
  1143→          if (epoch !== this.loadEpoch) return;
  1144→          if (error.name === "NotAllowedError") {
  1145→            log("yellow", "audio", "play blocked — waiting for user gesture");
  1146→            usePlayerStore.getState().setStatus("paused");
  1147→          } else if (error.name !== "AbortError") {
  1148→            log("red", "audio", `play failed: ${String(error)}`);
  1149→            if (currentTrack && this.recoveryAttempts < 1) {
  1150→              this.recoveryAttempts++;
  1151→              log(
  1152→                "yellow",
  1153→                "playback",
  1154→                "resume failed — attempting cache/stream recovery",
  1155→              );
  1156→              void this.loadAndPlay(
  1157→                currentTrack,
  1158→                usePlayerStore.getState().positionMs,
  1159→                true,
  1160→              );
  1161→              return;
  1162→            }
  1163→            usePlayerStore.getState().setStatus("paused");
  1164→          }
  1165→        })
  1166→        .finally(() => {
  1167→          if (epoch === this.loadEpoch) this.activePlayPromise = null;
  1168→        });
  1169→    } else {
  1170→      this.fadeVolume(targetVolume, 180);

  1195→  private async downloadAudioBlob(
  1196→    stream: PlaybackStream,
  1197→    signal: AbortSignal,
  1198→  ): Promise<Blob | null> {
  1199→    if (stream.blob) return stream.blob;
  1200→    if (stream.completed) return stream.completed;
  1201→    const startedAt = performance.now();
  1202→    let bytes: Awaited<ReturnType<typeof downloadAudioBytes>>;
  1203→    try {
  1204→      bytes = await downloadAudioBytes(stream.url, {
  1205→        contentLength: stream.contentLength,
  1206→        signal,
  1207→      });
  1208→    } catch (error) {
  1209→      if (stream.fromLiner && !signal.aborted)
  1210→        markLinerStreamFailed(`download: ${String(error)}`);
  1211→      throw error;
  1212→    }
  1213→    if (!bytes || signal.aborted) return null;
  1214→    const mimeType = stream.mimeType || "audio/webm";
  1215→    if (!mimeType.startsWith("audio/") && !mimeType.startsWith("video/"))
  1216→      return null;
  1217→    if (!isStandaloneAudioFile(bytes.subarray(0, 16))) {
  1218→      log(
  1219→        "yellow",
  1220→        "download",
  1221→        `not caching: stream is not a standalone audio file`,
  1222→      );
  1223→      return null;
  1224→    }
  1225→    log(
  1226→      "cyan",
  1227→      "download",
  1228→      `${Math.round(bytes.byteLength / 1024)} KB in ${Math.round(performance.now() - startedAt)}ms (${stream.contentLength ? "chunked" : "single request"})`,
  1229→    );
  1230→    return new Blob([bytes], { type: mimeType });
  1231→  }

  1233→  private cacheAudioInBackground(
  1234→    track: Track,
  1235→    stream: PlaybackStream,
  1236→    signal: AbortSignal,
  1237→  ) {
  1238→    if (track.localFileId) return; // never duplicate the user's own files
  1239→    if (track.durationMs && track.durationMs > 20 * 60 * 1000) return;
  1240→    const generation = audioCache.currentGeneration();
  1241→    void (async () => {
  1242→      try {
  1243→        const blob = await this.downloadAudioBlob(stream, signal);
  1244→        if (!blob || signal.aborted) return;
  1245→        if (
  1246→          !(await audioCache.saveAudioBlob(
  1247→            track.id,
  1248→            blob,
  1249→            blob.type,
  1250→            generation,
  1251→          ))
  1252→        )
  1253→          return;
  1254→        await linerDb.putTrack(track);
  1255→        log(
  1256→          "green",
  1257→          "playback",
  1258→          `background cached: ${track.title} (${Math.round(blob.size / 1024)} KB)`,
  1259→        );
  1260→      } catch {}
  1261→    })();
  1262→  }
